   136→export default function OwnerDashboardPage() {
   137→  const router = useRouter();
   138→  const { toast } = useToast();
   139→  const { data: session, status } = useSession();
   140→
   141→  const [listings, setListings] = useState<Listing[]>([]);
   142→  const [loadingListings, setLoadingListings] = useState(true);
   143→  const [formOpen, setFormOpen] = useState(false);
   144→  const [editing, setEditing] = useState<Listing | null>(null);
   145→  const [deleteId, setDeleteId] = useState<string | null>(null);
   146→  const [deleting, setDeleting] = useState(false);
   147→
   148→  // Beta status (client-side fetch)
   149→  const [betaStatus, setBetaStatus] = useState<BetaStatus | null>(null);
   150→
   151→  // Redirect na prijavo če ni prijavljen
   152→  useEffect(() => {
   153→    if (status === "unauthenticated") {
   154→      router.push("/owner/prijava");
   155→    }
   156→  }, [status, router]);
   157→
   158→  // Fetch beta status
   159→  useEffect(() => {
   160→    fetch("/api/beta-status")
   161→      .then((r) => r.json())
   162→      .then((d: BetaStatus) => setBetaStatus(d))
   163→      .catch(() => {});
   164→  }, []);
   165→
   166→  const fetchListings = useCallback(async () => {
   167→    setLoadingListings(true);
   168→    try {
   169→      const res = await fetch("/api/owner/listings", { cache: "no-store" });
   170→      if (!res.ok) throw new Error();
   171→      const data = await res.json();
   172→      setListings(data.listings || []);
   173→    } catch {
   174→      toast({
   175→        variant: "destructive",
   176→        title: "Napaka",
   177→        description: "Ni mogoče naložiti lokalov.",
   178→      });
   179→    } finally {
   180→      setLoadingListings(false);
   181→    }
   182→  }, [toast]);
   183→
   184→  useEffect(() => {
   185→    if (status === "authenticated") {
   186→      fetchListings();
   187→    }
   188→  }, [status, fetchListings]);
   189→
   190→  const plan = (session?.user?.plan as ListingPlan) || "free";
   191→  const isBetaActive = betaStatus?.isActive ?? true;
   192→  const planLimits = isBetaActive ? PLAN_LIMITS_BETA : PLAN_LIMITS_NORMAL;
   193→  const planLimit = planLimits[plan];
   194→  const planLimitLabel =
   195→    planLimit === Infinity ? "neomejeno" : String(planLimit);
   196→  const listingsCount = listings.length;
   197→  const canAddMore =
   198→    planLimit === Infinity ? true : listingsCount < planLimit;
   199→
   200→  const handleAdd = () => {
   201→    if (!canAddMore) {
   202→      toast({
   203→        variant: "destructive",
   204→        title: "Dosežen limit",
   205→        description: `Vaš paket (${plan}) omogoča največ ${planLimitLabel} ${
   206→          planLimit === 1 ? "lokal" : "lokalov"
   207→        }. Nadgradite naročnino.`,
   208→      });
   209→      return;
   210→    }
   211→    setEditing(null);
   212→    setFormOpen(true);
   213→  };
   214→
   215→  const handleEdit = (listing: Listing) => {
   216→    setEditing(listing);
   217→    setFormOpen(true);
   218→  };
   219→
   220→  const handleDelete = async () => {
   221→    if (!deleteId) return;
   222→    setDeleting(true);
   223→    try {
   224→      const res = await fetch(`/api/owner/listings/${deleteId}`, {
   225→        method: "DELETE",
   226→      });
   227→      if (!res.ok) {
   228→        const data = await res.json().catch(() => ({}));
   229→        throw new Error(data?.error ?? "Brisanje ni uspelo.");
   230→      }
   231→      toast({
   232→        title: "Izbrisano",
   233→        description: "Lokal je bil uspešno izbrisan.",
   234→      });
   235→      setListings((prev) => prev.filter((l) => l.id !== deleteId));
   236→    } catch (err) {
   237→      toast({
   238→        variant: "destructive",
   239→        title: "Napaka",
   240→        description:
   241→          err instanceof Error ? err.message : "Brisanje ni uspelo.",
   242→      });
   243→    } finally {
   244→      setDeleting(false);
   245→      setDeleteId(null);
   246→    }
   247→  };
   248→
   249→  // Loading state
   250→  if (status === "loading") {
   251→    return (
   252→      <main className="min-h-screen flex items-center justify-center bg-muted/30">
   253→        <div className="flex flex-col items-center gap-3 text-muted-foreground">
   254→          <Loader2 className="size-8 animate-spin" aria-hidden="true" />
   255→          <p className="text-sm">Nalagam portal...</p>
   256→        </div>
   257→      </main>
   258→    );
   259→  }
   260→
   261→  // Unauthenticated — redirect se sproži v useEffect
   262→  if (status === "unauthenticated" || !session?.user) {
   263→    return null;
   264→  }
   265→
   266→  return (
   267→    <main className="min-h-screen bg-muted/30 flex flex-col">
   268→      {/* Beta banner na vrhu */}
   269→      <BetaBanner />
   270→
   271→      {/* Header */}
   272→      <header className="bg-background border-b border-border sticky top-0 z-30">
   273→        <div className="mx-auto max-w-6xl px-4 py-3 sm:py-4 flex items-center justify-between gap-3">
   274→          <div className="flex items-center gap-3 min-w-0">
   275→            <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-primary text-primary-foreground">
   276→              <Building2 className="size-5" aria-hidden="true" />
   277→            </div>
   278→            <div className="min-w-0">
   279→              <h1 className="text-base font-bold sm:text-lg leading-tight">
   280→                Moj portal
   281→              </h1>
   282→              <p className="text-xs text-muted-foreground truncate">
   283→                {session.user.businessName || session.user.name}
   284→              </p>
   285→            </div>
   286→          </div>
   287→          <div className="flex items-center gap-2 shrink-0">
   288→            <PlanBadge plan={plan} />
   289→            <Button
   290→              variant="outline"
   291→              size="sm"
   292→              onClick={() => signOut({ callbackUrl: "/owner/prijava" })}
   293→              className="gap-1.5"
   294→            >
   295→              <LogOut className="size-3.5" aria-hidden="true" />
   296→              <span className="hidden sm:inline">Odjava</span>
   297→            </Button>
   298→          </div>
   299→        </div>
   300→      </header>
   301→
   302→      {/* Content */}
   303→      <div className="mx-auto max-w-6xl w-full px-4 py-6 sm:py-8 flex-1">
   304→        <Tabs defaultValue="listings" className="w-full">
   305→          <TabsList className="grid w-full grid-cols-3 sm:grid-cols-5 mb-6 gap-1">
   306→            <TabsTrigger value="listings" className="gap-1.5 text-xs sm:text-sm">
   307→              <Building className="size-3.5" aria-hidden="true" />
   308→              <span className="hidden sm:inline">Moji lokalci</span>
   309→              <span className="sm:hidden">Lokalci</span>
   310→            </TabsTrigger>
   311→            <TabsTrigger value="products" className="gap-1.5 text-xs sm:text-sm">
   312→              <Package className="size-3.5" aria-hidden="true" />
   313→              <span className="hidden sm:inline">Izdelki</span>
   314→              <span className="sm:hidden">Izdelki</span>
   315→            </TabsTrigger>
   316→            <TabsTrigger value="experiences" className="gap-1.5 text-xs sm:text-sm">
   317→              <Ticket className="size-3.5" aria-hidden="true" />
   318→              <span className="hidden sm:inline">Izkušnje</span>
   319→              <span className="sm:hidden">Izkušnje</span>
   320→            </TabsTrigger>
   321→            <TabsTrigger value="narocnina" className="gap-1.5 text-xs sm:text-sm">
   322→              <Crown className="size-3.5" aria-hidden="true" />

   327→              <TrendingUp className="size-3.5" aria-hidden="true" />
   328→              <span className="hidden sm:inline">Statistika</span>

   332→
   333→          {/* TAB 1: Moji lokalci */}
   334→          <TabsContent value="listings" className="space-y-4">

   336→              <div>
   337→                <h2 className="text-lg font-bold sm:text-xl">
   338→                  Moji lokalci
   339→                </h2>
   340→                <p className="text-sm text-muted-foreground">
   341→                  {listingsCount} od {planLimitLabel} lokalov · paket{" "}

   345→              <Button

   351→                Dodaj lokal

   353→            </div>
   354→
   355→            {/* Beta info badge v listings tab */}
   356→            {isBetaActive && (
   357→              <div className="flex flex-col sm:flex-row sm:items-center gap-3 rounded-lg border border-amber-300/60 bg-amber-50 dark:bg-amber-950/20 p-4">
   358→                <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300">
   359→                  <Rocket className="size-5" aria-hidden="true" />
   360→                </div>
   361→                <div className="flex-1 min-w-0">
   362→                  <p className="text-sm font-semibold text-amber-900 dark:text-amber-200">
   363→                    Beta: {plan === "free" ? "3 lokalci brezplačno" : plan === "premium" ? "8 lokalcev brezplačno" : "neomejeno brezplačno"}
   364→                    {plan === "free" && " (običajno 1)"}
   365→                    {plan === "premium" && " (običajno 5)"}
   366→                  </p>
   367→                  <p className="text-xs text-amber-700/80 dark:text-amber-300/70 mt-0.5">
   368→                    Med beta obdobjem so vsi paketi brezplačni. Vaš paket{" "}
   369→                    {PLAN_LABELS[plan]} vam omogoča {planLimitLabel} lokalov.
   370→                  </p>
   371→                </div>
   372→              </div>
   373→            )}
   374→
   375→            {!canAddMore && (
   376→              <Alert className="border-amber-400/50 bg-amber-50 dark:bg-amber-950/20">
   377→                <Crown className="size-4 text-amber-600" aria-hidden="true" />
   378→                <AlertTitle>Dosežen limit paketa</AlertTitle>
   379→                <AlertDescription>
   380→                  Vaš paket ({PLAN_LABELS[plan]}) omogoča največ{" "}
   381→                  {planLimitLabel}{" "}
   382→                  {planLimit === 1 ? "lokal" : "lokalov"}. Za več lokalov
   383→                  nadgradite na višji paket v zavihku &laquo;Naročnina&raquo;.
   384→                </AlertDescription>
   385→              </Alert>
   386→            )}
   387→
   388→            {loadingListings ? (
   389→              <div className="flex items-center justify-center py-16">
   390→                <Loader2
   391→                  className="size-8 animate-spin text-muted-foreground"
   392→                  aria-hidden="true"
   393→                />
   394→              </div>
   395→            ) : listings.length === 0 ? (
   396→              <EmptyState onAdd={handleAdd} canAdd={canAddMore} />
   397→            ) : (
   398→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   399→                {listings.map((listing) => (
   400→                  <ListingCard
   401→                    key={listing.id}
   402→                    listing={listing}
   403→                    onEdit={() => handleEdit(listing)}
   404→                    onDelete={() => setDeleteId(listing.id)}
   405→                  />
   406→                ))}
   407→              </div>
   408→            )}
   409→          </TabsContent>
   410→
   411→          {/* TAB 2: Izdelki */}
   412→          <TabsContent value="products" className="space-y-4">
   413→            <ProductsTab plan={plan} isBetaActive={isBetaActive} />
   414→          </TabsContent>
   415→
   416→          {/* TAB 3: Izkušnje */}
   417→          <TabsContent value="experiences" className="space-y-4">
   418→            <ExperiencesTab plan={plan} isBetaActive={isBetaActive} />
   419→          </TabsContent>
   420→
   421→          {/* TAB 4: Naročnina */}
   422→          <TabsContent value="narocnina" className="space-y-6">
   423→            <SubscriptionTab
   424→              plan={plan}
   425→              session={session}
   426→              betaStatus={betaStatus}
   427→              onUpgraded={fetchListings}
   428→            />
   429→          </TabsContent>
   430→
   431→          {/* TAB 5: Statistika */}
   432→          <TabsContent value="statistika" className="space-y-6">
   433→            <StatisticsTab listings={listings} loading={loadingListings} />
   434→          </TabsContent>
   435→        </Tabs>
   436→      </div>
   437→
   438→      {/* Listing form dialog */}
   439→      <ListingFormDialog
   440→        open={formOpen}
   441→        onOpenChange={setFormOpen}
   442→        listing={editing}
   443→        onSaved={fetchListings}
   444→      />
   445→
   446→      {/* Delete confirmation */}
   447→      <AlertDialog
   448→        open={deleteId !== null}
   449→        onOpenChange={(open) => !open && setDeleteId(null)}
   450→      >
   451→        <AlertDialogContent>
   452→          <AlertDialogHeader>
   453→            <AlertDialogTitle>Izbriši lokal?</AlertDialogTitle>
   454→            <AlertDialogDescription>
   455→              To dejanje je nepovratno. Lokal bo trajno odstranjen iz portala
   456→              in ne bo več prikazan obiskovalcem.
   457→            </AlertDialogDescription>
   458→          </AlertDialogHeader>
   459→          <AlertDialogFooter>
   460→            <AlertDialogCancel disabled={deleting}>Prekliči</AlertDialogCancel>
   461→            <AlertDialogAction
   462→              onClick={handleDelete}
   463→              disabled={deleting}
   464→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90 gap-1.5"
   465→            >
   466→              {deleting ? (
   467→                <>
   468→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   469→                  Brišem...
   470→                </>
   471→              ) : (
   472→                <>
   473→                  <Trash2 className="size-4" aria-hidden="true" />
   474→                  Izbriši
   475→                </>
   476→              )}
   477→            </AlertDialogAction>
   478→          </AlertDialogFooter>
   479→        </AlertDialogContent>
   480→      </AlertDialog>
   481→    </main>
   482→  );
   483→}

   486→
   487→function PlanBadge({ plan }: { plan: ListingPlan }) {
   488→  if (plan === "premium") {
   489→    return (
   490→      <Badge className="bg-amber-400 text-amber-950 hover:bg-amber-400 border-0 gap-1">
   491→        <Star className="size-3 fill-amber-950" aria-hidden="true" />
   492→        <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   493→      </Badge>
   494→    );
   495→  }
   496→  if (plan === "enterprise") {
   497→    return (
   498→      <Badge className="bg-primary text-primary-foreground hover:bg-primary border-0 gap-1">
   499→        <Crown className="size-3" aria-hidden="true" />
   500→        <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   501→      </Badge>
   502→    );
   503→  }
   504→  return (
   505→    <Badge variant="secondary" className="gap-1">
   506→      <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   507→      <span className="sm:hidden">Free</span>
   508→    </Badge>
   509→  );
   510→}
