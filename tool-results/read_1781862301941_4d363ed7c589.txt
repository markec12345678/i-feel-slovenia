   176→  const [listings, setListings] = React.useState<AdminListing[]>([]);
   177→  const [loading, setLoading] = React.useState(true);
   178→  const [errorMsg, setErrorMsg] = React.useState<string | null>(null);
   179→  const [search, setSearch] = React.useState("");
   180→
   181→  const [formOpen, setFormOpen] = React.useState(false);
   182→  const [editing, setEditing] = React.useState<AdminListing | null>(null);
   183→  const [deleteId, setDeleteId] = React.useState<string | null>(null);
   184→  const [deleting, setDeleting] = React.useState(false);
   185→
   186→  const fetchListings = React.useCallback(async () => {
   187→    setLoading(true);
   188→    setErrorMsg(null);
   189→    try {
   190→      const res = await fetch("/api/admin/listings", {
   191→        headers: { "x-admin-password": adminPassword },
   192→      });
   193→      if (!res.ok) {
   194→        const d: unknown = await res.json();
   195→        const msg =
   196→          typeof d === "object" && d !== null && "error" in d
   197→            ? String((d as Record<string, unknown>).error)
   198→            : "Napaka pri pridobivanju lokalov";
   199→        setErrorMsg(msg);
   200→        return;
   201→      }
   202→      const data: unknown = await res.json();
   203→      const list = (
   204→        typeof data === "object" && data !== null && "listings" in data
   205→          ? (data as Record<string, unknown>).listings
   206→          : []) as AdminListing[];
   207→      setListings(list);
   208→    } catch (err) {
   209→      console.error("[admin/listings] fetch:", err);
   210→      setErrorMsg("Napaka pri povezavi s strežnikom");
   211→    } finally {
   212→      setLoading(false);
   213→    }
   214→  }, [adminPassword]);
   215→
   216→  React.useEffect(() => {
   217→    fetchListings();
   218→  }, [fetchListings]);
   219→
   220→  const filtered = React.useMemo(() => {
   221→    if (!search.trim()) return listings;
   222→    const q = search.toLowerCase();
   223→    return listings.filter(
   224→      (l) =>
   225→        l.name.toLowerCase().includes(q) ||
   226→        (l.destinationName ?? "").toLowerCase().includes(q) ||
   227→        l.category.toLowerCase().includes(q)
   228→    );
   229→  }, [listings, search]);
   230→
   231→  const handleOpenCreate = () => {
   232→    setEditing(null);
   233→    setFormOpen(true);
   234→  };
   235→
   236→  const handleOpenEdit = (listing: AdminListing) => {
   237→    setEditing(listing);
   238→    setFormOpen(true);
   239→  };
   240→
   241→  const handleSaved = () => {
   242→    fetchListings();
   243→  };
   244→
   245→  const handleConfirmDelete = async () => {
   246→    if (!deleteId) return;
   247→    setDeleting(true);
   248→    try {
   249→      const res = await fetch(`/api/admin/listings/${deleteId}`, {
   250→        method: "DELETE",
   251→        headers: { "x-admin-password": adminPassword },
   252→      });

   254→        const d: unknown = await res.json();
   255→        const msg =
   256→          typeof d === "object" && d !== null && "error" in d
   257→            ? String((d as Record<string, unknown>).error)
   258→            : "Napaka pri brisanju";
   259→        setErrorMsg(msg);
   260→      } else {
   261→        setListings((prev) => prev.filter((l) => l.id !== deleteId));
   262→      }
   263→    } catch (err) {
   264→      console.error("[admin/listings] delete:", err);
   265→      setErrorMsg("Napaka pri povezavi s strežnikom");
   266→    } finally {
   267→      setDeleting(false);
   268→      setDeleteId(null);
   269→    }

   271→
   272→  return (
   273→    <div className="space-y-4">
   274→      {/* Toolbar */}
   275→      <div className="flex flex-col sm:flex-row gap-3 sm:items-center sm:justify-between">

   277→          <h2 className="text-xl font-bold tracking-tight">Lokali</h2>

   285→            <Input
   286→              value={search}
   287→              onChange={(e) => setSearch(e.target.value)}
   288→              placeholder="Iskanje po imenu..."
   289→              className="pl-8 sm:w-64"
   290→            />
   291→          </div>
   292→          <Button onClick={handleOpenCreate} className="shrink-0">
   293→            <Plus className="size-4" />

   470→function LeadsTab({ adminPassword }: { adminPassword: string }) {
   471→  const [leads, setLeads] = React.useState<Lead[]>([]);
   472→  const [loading, setLoading] = React.useState(true);
   473→  const [errorMsg, setErrorMsg] = React.useState<string | null>(null);
   474→
   475→  const fetchLeads = React.useCallback(async () => {
   476→    setLoading(true);
   477→    setErrorMsg(null);
   478→    try {
   479→      const res = await fetch("/api/admin/leads", {
   480→        headers: { "x-admin-password": adminPassword },
   481→      });
   482→      if (!res.ok) {
   483→        const d: unknown = await res.json();
   484→        const msg =
   485→          typeof d === "object" && d !== null && "error" in d
   486→            ? String((d as Record<string, unknown>).error)
   487→            : "Napaka pri pridobivanju leadov";
   488→        setErrorMsg(msg);
   489→        return;
   490→      }
   491→      const data: unknown = await res.json();
   492→      const list = (
   493→        typeof data === "object" && data !== null && "leads" in data
   494→          ? (data as Record<string, unknown>).leads
   495→          : []) as Lead[];
   496→      setLeads(list);
   497→    } catch (err) {
   498→      console.error("[admin/leads] fetch:", err);
   499→      setErrorMsg("Napaka pri povezavi s strežnikom");
   500→    } finally {
   501→      setLoading(false);
   502→    }
   503→  }, [adminPassword]);
   504→
   505→  React.useEffect(() => {
   506→    fetchLeads();
   507→  }, [fetchLeads]);
   508→
   509→  const handleCycleStatus = async (lead: Lead) => {
   510→    const next = STATUS_NEXT[lead.status];
   511→    // Optimistic update
   512→    setLeads((prev) =>
   513→      prev.map((l) => (l.id === lead.id ? { ...l, status: next } : l))
   514→    );
   515→    try {
   516→      const res = await fetch("/api/admin/leads", {
   517→        method: "PUT",
   518→        headers: {
   519→          "Content-Type": "application/json",
   520→          "x-admin-password": adminPassword,
   521→        },
   522→        body: JSON.stringify({ id: lead.id, status: next }),
   523→      });
   524→      if (!res.ok) {
   525→        // Revert on failure
   526→        setLeads((prev) =>
   527→          prev.map((l) => (l.id === lead.id ? { ...l, status: lead.status } : l))
   528→        );
   529→        const d: unknown = await res.json();
   530→        const msg =
   531→          typeof d === "object" && d !== null && "error" in d
   532→            ? String((d as Record<string, unknown>).error)
   533→            : "Napaka pri posodabljanju";
   534→        setErrorMsg(msg);
   535→      }
   536→    } catch (err) {
   537→      console.error("[admin/leads] update:", err);
   538→      setLeads((prev) =>
   539→        prev.map((l) => (l.id === lead.id ? { ...l, status: lead.status } : l))
   540→      );
   541→      setErrorMsg("Napaka pri povezavi s strežnikom");
   542→    }
   543→  };
   544→
   545→  const handleExportCsv = () => {
   546→    if (leads.length === 0) return;
   547→    const headers = [
   548→      "Datum",
   549→      "Ime",
   550→      "Email",
   551→      "Telefon",
   552→      "Lokal",
   553→      "Tip",
   554→      "Kraj",
   555→      "Paket",
   556→      "Status",
   557→      "Sporočilo",
   558→    ];
   559→    const rows = leads.map((l) => [
   560→      new Date(l.timestamp).toLocaleString("sl-SI"),
   561→      l.name,
   562→      l.email,
   563→      l.phone ?? "",
   564→      l.businessName,
   565→      l.businessType,
   566→      l.location,
   567→      l.plan,
   568→      STATUS_LABELS[l.status],
   569→      (l.message ?? "").replace(/[\r\n]+/g, " "),
   570→    ]);
   571→    const csv = [headers, ...rows]
   572→      .map((row) =>
   573→        row
   574→          .map((cell) => {
   575→            const s = String(cell ?? "");
   576→            if (s.includes(";") || s.includes('"') || s.includes("\n")) {
   577→              return `"${s.replace(/"/g, '""')}"`;
   578→            }
   579→            return s;
   580→          })
   581→          .join(";")
   582→      )
   583→      .join("\n");
   584→
   585→    // BOM za Excel UTF-8

   721→function StatsTab({ adminPassword }: { adminPassword: string }) {
   722→  const [listings, setListings] = React.useState<AdminListing[]>([]);
   723→  const [leads, setLeads] = React.useState<Lead[]>([]);
   724→  const [loading, setLoading] = React.useState(true);
   725→
   726→  React.useEffect(() => {
   727→    let cancelled = false;
   728→    (async () => {
   729→      setLoading(true);
   730→      try {
   731→        const [lRes, leadsRes] = await Promise.all([
   732→          fetch("/api/admin/listings", {
   733→            headers: { "x-admin-password": adminPassword },
   734→          }),
   735→          fetch("/api/admin/leads", {
   736→            headers: { "x-admin-password": adminPassword },
   737→          }),
   738→        ]);
   739→        const lData: unknown = await lRes.json();
   740→        const leadsData: unknown = await leadsRes.json();
   741→        if (cancelled) return;
   742→        const lList = (
   743→          typeof lData === "object" && lData !== null && "listings" in lData
   744→            ? (lData as Record<string, unknown>).listings
   745→            : []) as AdminListing[];
   746→        const leadsList = (
   747→          typeof leadsData === "object" && leadsData !== null && "leads" in leadsData
   748→            ? (leadsData as Record<string, unknown>).leads
   749→            : []) as Lead[];
   750→        setListings(lList);
   751→        setLeads(leadsList);
   752→      } catch (err) {
   753→        console.error("[admin/stats] fetch:", err);
   754→      } finally {
   755→        if (!cancelled) setLoading(false);
   756→      }
   757→    })();
   758→    return () => {
   759→      cancelled = true;
   760→    };
   761→  }, [adminPassword]);
   762→
   763→  // Kategorije — useMemo MORA biti pred morebitnim zgodnjim returnom
   764→  const byCategory = React.useMemo(() => {
   765→    const map = new Map<string, number>();
   766→    for (const l of listings) {
   767→      map.set(l.category, (map.get(l.category) ?? 0) + 1);
   768→    }
   769→    const arr = Array.from(map.entries()).map(([cat, count]) => ({
   770→      cat: cat as ListingCategory,
   771→      label: CATEGORY_LABELS[cat as ListingCategory] ?? cat,
   772→      count,
   773→    }));
   774→    arr.sort((a, b) => b.count - a.count);
   775→    return arr;
   776→  }, [listings]);
   777→
   778→  if (loading) {
   779→    return (
   780→      <div className="flex items-center justify-center gap-2 py-16 text-muted-foreground">
   781→        <Loader2 className="size-4 animate-spin" />

   783→      </div>
   784→    );
   785→  }
   786→
   787→  const total = listings.length;
   788→  const premium = listings.filter(
   789→    (l) => l.plan === "premium" || l.plan === "enterprise"
   790→  ).length;
   791→  const totalLeads = leads.length;
   792→  const totalViews = listings.reduce((sum, l) => sum + (l.viewCount ?? 0), 0);
   793→
   794→  const top5 = [...listings]
   795→    .sort((a, b) => (b.viewCount ?? 0) - (a.viewCount ?? 0))
   796→    .slice(0, 5);

   803→        <h2 className="text-xl font-bold tracking-tight">Statistika</h2>

   805→          Pregled ključnih metrik platforme.

   812→          icon={<Building2 className="size-5" />}
   813→          label="Skupno lokalov"

   817→        <KpiCard

   820→          value={premium}

   827→          color="accent"

   957→function KpiCard({
   958→  icon,
   959→  label,
   960→  value,

   979→            <p className="text-xs sm:text-sm text-muted-foreground truncate">
