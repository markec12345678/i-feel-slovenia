   102→/**
   103→ * MarketplaceSection — tržnica slovenskih izdelkov in izkušenj.
   104→ * Tabs med izdelki in izkušnjami, filtri kategorije in sortiranja.
   105→ * Kartice odprejo detail modal (ProductModal / ExperienceModal).
   106→ */
   107→export function MarketplaceSection() {
   108→  const [tab, setTab] = useState<Tab>("products");
   109→
   110→  // Filtri izdelki
   111→  const [productCategory, setProductCategory] = useState<string>(ALL_VALUE);
   112→  const [productSort, setProductSort] = useState<string>("featured");
   113→
   114→  // Filtri izkušnje
   115→  const [expCategory, setExpCategory] = useState<string>(ALL_VALUE);
   116→  const [expSort, setExpSort] = useState<string>("featured");
   117→
   118→  const [products, setProducts] = useState<Product[]>([]);
   119→  const [productsTotal, setProductsTotal] = useState<number>(0);
   120→  const [productsLoading, setProductsLoading] = useState<boolean>(true);
   121→  const [productsError, setProductsError] = useState<string | null>(null);
   122→
   123→  const [experiences, setExperiences] = useState<Experience[]>([]);
   124→  const [experiencesTotal, setExperiencesTotal] = useState<number>(0);
   125→  const [experiencesLoading, setExperiencesLoading] = useState<boolean>(true);
   126→  const [experiencesError, setExperiencesError] = useState<string | null>(null);
   127→
   128→  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null);
   129→  const [selectedExperience, setSelectedExperience] =
   130→    useState<Experience | null>(null);
   131→
   132→  // Fetch izdelkov
   133→  const fetchProducts = useCallback(async () => {
   134→    setProductsLoading(true);
   135→    setProductsError(null);
   136→    try {
   137→      const params = new URLSearchParams();
   138→      if (productCategory !== ALL_VALUE)
   139→        params.set("category", productCategory);
   140→      params.set("sort", productSort);
   141→      params.set("limit", "50");
   142→
   143→      const res = await fetch(`/api/products?${params.toString()}`, {
   144→        cache: "no-store",
   145→      });
   146→      if (!res.ok) throw new Error("Napaka pri pridobivanju izdelkov");
   147→      const data: ProductsResponse = await res.json();
   148→      setProducts(data.products ?? []);
   149→      setProductsTotal(data.total ?? 0);
   150→    } catch (err) {
   151→      console.error("[products] fetch napaka:", err);
   152→      setProductsError("Ne morem naložiti izdelkov. Poskusite kasneje.");
   153→      setProducts([]);
   154→      setProductsTotal(0);
   155→    } finally {
   156→      setProductsLoading(false);
   157→    }
   158→  }, [productCategory, productSort]);
   159→
   160→  // Fetch izkušenj
   161→  const fetchExperiences = useCallback(async () => {
   162→    setExperiencesLoading(true);
   163→    setExperiencesError(null);
   164→    try {
   165→      const params = new URLSearchParams();
   166→      if (expCategory !== ALL_VALUE) params.set("category", expCategory);
   167→      params.set("sort", expSort);
   168→      params.set("limit", "50");
   169→
   170→      const res = await fetch(`/api/experiences?${params.toString()}`, {
   171→        cache: "no-store",
   172→      });
   173→      if (!res.ok) throw new Error("Napaka pri pridobivanju izkušenj");
   174→      const data: ExperiencesResponse = await res.json();
   175→      setExperiences(data.experiences ?? []);
   176→      setExperiencesTotal(data.total ?? 0);
   177→    } catch (err) {
   178→      console.error("[experiences] fetch napaka:", err);
   179→      setExperiencesError("Ne morem naložiti izkušenj. Poskusite kasneje.");
   180→      setExperiences([]);
   181→      setExperiencesTotal(0);
   182→    } finally {
   183→      setExperiencesLoading(false);
   184→    }
   185→  }, [expCategory, expSort]);
   186→
   187→  useEffect(() => {
   188→    void fetchProducts();
   189→  }, [fetchProducts]);
   190→
   191→  useEffect(() => {
   192→    void fetchExperiences();
   193→  }, [fetchExperiences]);
   194→
   195→  // Počisti filtre glede na aktivni tab
   196→  const clearFilters = () => {
   197→    if (tab === "products") {
   198→      setProductCategory(ALL_VALUE);
   199→      setProductSort("featured");
   200→    } else {
   201→      setExpCategory(ALL_VALUE);
   202→      setExpSort("featured");
   203→    }
   204→  };
   205→
   206→  const hasActiveFilters =
   207→    tab === "products"
   208→      ? productCategory !== ALL_VALUE || productSort !== "featured"
   209→      : expCategory !== ALL_VALUE || expSort !== "featured";
   210→
   211→  return (
   212→    <section
   213→      id="trznica"
   214→      className="scroll-mt-20 bg-muted/20 py-16 sm:py-20"
   215→      aria-labelledby="trznica-title"
   216→    >
   217→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   218→        {/* Header */}
   219→        <div className="mx-auto max-w-2xl text-center">
   220→          <Badge
   221→            variant="secondary"
   222→            className="mb-3 gap-1.5 bg-primary/10 text-primary"
   223→          >
   224→            <Store className="size-3.5" aria-hidden="true" />
   225→            Tržnica
   226→          </Badge>
   227→          <h2
   228→            id="trznica-title"
   229→            className="text-3xl font-bold tracking-tight sm:text-4xl"
   230→          >
   231→            Tržnica Slovenije
   232→          </h2>
   233→          <p className="mt-3 text-base text-muted-foreground">
   234→            Lokalni izdelki in izkušnje — direktno od kmetov, vinogradnikov in
   235→            vodnikov
   236→          </p>
   237→        </div>
   238→
   239→        {/* Tabs */}
   240→        <div className="mt-8 flex justify-center">
   241→          <Tabs
   242→            value={tab}
   243→            onValueChange={(v) => setTab(v as Tab)}
   244→            className="w-full max-w-md"
   245→          >
   246→            <TabsList className="grid w-full grid-cols-2">
   247→              <TabsTrigger value="products" className="gap-1.5">
   248→                <ShoppingBag className="size-4" aria-hidden="true" />
   249→                Izdelki
   250→              </TabsTrigger>
   251→              <TabsTrigger value="experiences" className="gap-1.5">
   252→                <Compass className="size-4" aria-hidden="true" />
   253→                Izkušnje
   254→              </TabsTrigger>
   255→            </TabsList>
   256→          </Tabs>
   257→        </div>
   258→
   259→        {/* Filter vrstica */}
   260→        <div className="mx-auto mt-6 max-w-3xl rounded-xl border border-border/60 bg-background p-4 sm:p-5">
   261→          <div className="mb-3 flex items-center justify-between gap-2">
   262→            <div className="flex items-center gap-2 text-sm font-medium text-foreground">
   263→              <Filter className="size-4 text-primary" aria-hidden="true" />
   264→              Filtri
   265→            </div>
   266→            {hasActiveFilters ? (
   267→              <Button
   268→                type="button"
   269→                variant="ghost"
   270→                size="sm"
   271→                onClick={clearFilters}
   272→                className="h-8 gap-1.5 text-muted-foreground hover:text-foreground"
   273→              >
   274→                <X className="size-3.5" aria-hidden="true" />
   275→                Počisti filtre
   276→              </Button>
   277→            ) : null}
   278→          </div>
   279→
   280→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   281→            {tab === "products" ? (
   282→              <>
   283→                <FilterSelect
   284→                  value={productCategory}
   285→                  onChange={setProductCategory}
   286→                  placeholder="Vse kategorije"
   287→                  ariaLabel="Filtriraj izdelke po kategoriji"
   288→                  options={PRODUCT_CATEGORY_OPTIONS}
   289→                />
   290→                <FilterSelect
   291→                  value={productSort}
   292→                  onChange={setProductSort}
   293→                  placeholder="Razvrsti po"
   294→                  ariaLabel="Razvrsti izdelke"
   295→                  options={PRODUCT_SORT_OPTIONS}
   296→                  showAllOption={false}
   297→                />
   298→              </>
   299→            ) : (
   300→              <>
   301→                <FilterSelect
   302→                  value={expCategory}
   303→                  onChange={setExpCategory}
   304→                  placeholder="Vse kategorije"
   305→                  ariaLabel="Filtriraj izkušnje po kategoriji"
   306→                  options={EXPERIENCE_CATEGORY_OPTIONS}
   307→                />
   308→                <FilterSelect
   309→                  value={expSort}
   310→                  onChange={setExpSort}
   311→                  placeholder="Razvrsti po"
   312→                  ariaLabel="Razvrsti izkušnje"
   313→                  options={EXPERIENCE_SORT_OPTIONS}
   314→                  showAllOption={false}
   315→                />
   316→              </>
   317→            )}
   318→          </div>
   319→        </div>
   320→
   321→        {/* Števec */}
   322→        <p className="mt-5 text-center text-sm text-muted-foreground">
   323→          {tab === "products"
   324→            ? productsLoading
   325→              ? "Nalagam izdelke..."
   326→              : (() => {
   327→                  const t = productsTotal;
   328→                  return (
   329→                    <>
   330→                      Prikazujem{" "}
   331→                      <span className="font-semibold text-foreground">{t}</span>{" "}
   332→                      {t === 1 ? "izdelek" : t < 5 ? "izdelke" : "izdelkov"}
   333→                    </>
   334→                  );
   335→                })()
   336→            : experiencesLoading
   337→              ? "Nalagam izkušnje..."
   338→              : (() => {
   339→                  const t = experiencesTotal;
   340→                  return (
   341→                    <>
   342→                      Prikazujem{" "}
   343→                      <span className="font-semibold text-foreground">{t}</span>{" "}
   344→                      {t === 1
   345→                        ? "izkušnjo"
   346→                        : t < 5
   347→                          ? "izkušnje"
   348→                          : "izkušenj"}
   349→                    </>
   350→                  );
   351→                })()}
   352→        </p>
   353→
   354→        {/* Grid */}
   355→        {tab === "products" ? (
   356→          productsLoading ? (
   357→            <div className="mt-6 grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
   358→              {Array.from({ length: 6 }).map((_, i) => (
   359→                <ProductSkeleton key={i} />
   360→              ))}
   361→            </div>
   362→          ) : productsError ? (
   363→            <ErrorState
   364→              message={productsError}
   365→              onRetry={() => void fetchProducts()}
   366→            />
   367→          ) : products.length === 0 ? (
   368→            <EmptyState
   369→              canClear={hasActiveFilters}
   370→              onClear={clearFilters}
   371→              label="izdelkov"
   372→            />
   373→          ) : (
   374→            <div className="mt-6 grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
   375→              {products.map((p) => (
   376→                <ProductCard
   377→                  key={p.id}
   378→                  product={p}
   379→                  onOpen={() => setSelectedProduct(p)}
   380→                />
   381→              ))}
   382→            </div>
   383→          )
   384→        ) : experiencesLoading ? (
   385→          <div className="mt-6 grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
   386→            {Array.from({ length: 6 }).map((_, i) => (
   387→              <ExperienceSkeleton key={i} />
   388→            ))}
   389→          </div>
   390→        ) : experiencesError ? (
   391→          <ErrorState
   392→            message={experiencesError}
   393→            onRetry={() => void fetchExperiences()}
   394→          />
   395→        ) : experiences.length === 0 ? (
   396→          <EmptyState
   397→            canClear={hasActiveFilters}
   398→            onClear={clearFilters}
   399→            label="izkušenj"
   400→          />
   401→        ) : (
   402→          <div className="mt-6 grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
   403→            {experiences.map((e) => (
   404→              <ExperienceCard
   405→                key={e.id}
   406→                experience={e}
   407→                onOpen={() => setSelectedExperience(e)}
   408→              />
   409→            ))}
   410→          </div>
   411→        )}
   412→
   413→        {/* Footer note — monetizacijski CTA */}
   414→        <div className="mt-10 flex flex-col items-center justify-center gap-2 rounded-xl border border-primary/30 bg-primary/5 p-6 text-center sm:flex-row sm:gap-4">
   415→          <Store className="size-5 text-primary" aria-hidden="true" />
   416→          <p className="text-sm text-foreground/90">
   417→            Želite prodajati svoje izdelke ali izkušnje? Pridruži se tržnici.
   418→          </p>
   419→          <a
   420→            href="#pridruzi-se"
   421→            className="inline-flex items-center gap-1.5 text-sm font-semibold text-primary transition-colors hover:text-primary/80"
   422→          >
   423→            Pridruži se
   424→            <Compass className="size-3.5" aria-hidden="true" />
   425→          </a>
   426→        </div>
   427→      </div>
   428→
   429→      {/* Modala */}
   430→      <ProductModal
   431→        product={selectedProduct}
   432→        onClose={() => setSelectedProduct(null)}
   433→      />
   434→      <ExperienceModal
   435→        experience={selectedExperience}
   436→        onClose={() => setSelectedExperience(null)}
   437→      />
   438→    </section>
   439→  );
   440→}
   441→
   442→interface FilterOption {
   443→  value: string;
   444→  label: string;
   445→}
   446→
   447→function FilterSelect({
   448→  value,
   449→  onChange,
   450→  placeholder,
   451→  ariaLabel,
   452→  options,
   453→  showAllOption = true,
   454→}: {
   455→  value: string;
   456→  onChange: (v: string) => void;
   457→  placeholder: string;
   458→  ariaLabel: string;
   459→  options: FilterOption[];
   460→  showAllOption?: boolean;
   461→}) {
   462→  return (
   463→    <Select value={value} onValueChange={onChange}>
   464→      <SelectTrigger aria-label={ariaLabel} className="w-full">
   465→        <SelectValue placeholder={placeholder} />
   466→      </SelectTrigger>
   467→      <SelectContent>
   468→        {showAllOption ? (
   469→          <SelectItem value={ALL_VALUE}>{placeholder}</SelectItem>
   470→        ) : null}
   471→        {options.map((o) => (
   472→          <SelectItem key={o.value} value={o.value}>
   473→            {o.label}
   474→          </SelectItem>
   475→        ))}
   476→      </SelectContent>
   477→    </Select>
   478→  );
   479→}

   481→/**
   482→ * ProductCard — kartica izdelka za tržnico.
   483→ */
   484→function ProductCard({
   485→  product,
   486→  onOpen,
   487→}: {
   488→  product: Product;
   489→  onOpen: () => void;
   490→}) {
   491→  const image = product.images[0];
   492→  const discount = product.compareAtPrice
   493→    ? Math.round(
   494→        ((product.compareAtPrice - product.price) / product.compareAtPrice) *
   495→          100
   496→      )
   497→    : 0;
   498→
   499→  // Preusmeritev na ponudnika — mi samo usmerjamo promet, ne pobiramo plačil
   500→  const handleVisitSeller = () => {
   501→    if (product.sellerWebsite) {
   502→      window.open(product.sellerWebsite, "_blank", "noopener,noreferrer");
   503→    }
   504→  };
   505→
   506→  return (
   507→    <Card
   508→      className={cn(
   509→        "group relative gap-0 overflow-hidden py-0 transition-all hover:shadow-lg focus-within:shadow-lg",
   510→        product.plan === "premium" && "border-primary"
   511→      )}
   512→    >
   513→      {/* Slika */}
   514→      <div className="relative aspect-square w-full overflow-hidden bg-muted">
   515→        {image ? (
   516→          <img
   517→            src={image}
   518→            alt={`${product.name} — ${product.description}`}
   519→            loading="lazy"
   520→            className="size-full object-cover transition-transform duration-500 group-hover:scale-105"
   521→          />
   522→        ) : (
   523→          <div className="flex size-full items-center justify-center bg-muted text-4xl">
   524→            <span aria-hidden="true">
   525→              {PRODUCT_CATEGORY_ICONS[product.category]}
   526→            </span>
   527→          </div>
   528→        )}
   529→
   530→        {/* Atributi (top-left) */}
   531→        <div className="absolute left-3 top-3 flex flex-col gap-1.5">
   532→          {product.organic ? (
   533→            <Badge className="bg-primary text-primary-foreground shadow-sm">
   534→              <Leaf className="size-3" aria-hidden="true" />
   535→              Ekološko
   536→            </Badge>
   537→          ) : null}
   538→          {product.handmade ? (
   539→            <Badge className="bg-blue-600 text-white shadow-sm">
   540→              <HandHeart className="size-3" aria-hidden="true" />
   541→              Ročno
   542→            </Badge>
   543→          ) : null}
   544→          {product.vegan ? (
   545→            <Badge variant="secondary" className="shadow-sm">
   546→              <Leaf className="size-3" aria-hidden="true" />
   547→              Vegansko
   548→            </Badge>
   549→          ) : null}
   550→        </div>
   551→
   552→        {/* Featured badge (top-right) */}
   553→        {product.featured ? (
   554→          <Badge className="absolute right-3 top-3 bg-amber-400 text-amber-950 shadow-sm">
   555→            <Sparkles className="size-3" aria-hidden="true" />
   556→            Izpostavljeno
   557→          </Badge>
   558→        ) : null}
   559→
   560→        {/* Discount badge (bottom-right) */}
   561→        {discount > 0 ? (
   562→          <Badge className="absolute bottom-3 right-3 bg-destructive text-destructive-foreground shadow-sm">
   563→            -{discount}%
   564→          </Badge>
   565→        ) : null}
   566→      </div>
   567→
   568→      {/* Body */}
   569→      <CardContent className="flex flex-col gap-3 p-4">
   570→        <div>
   571→          <h3 className="line-clamp-1 text-lg font-semibold leading-tight">
   572→            {product.name}
   573→          </h3>
   574→          <p className="mt-1 text-sm text-muted-foreground line-clamp-2">
   575→            {product.description}
   576→          </p>
   577→        </div>
   578→
   579→        {/* Rating */}
   580→        <div className="flex items-center gap-1.5">
   581→          <Star
   582→            className="size-4 fill-amber-400 text-amber-400"
   583→            aria-hidden="true"
   584→          />
   585→          <span className="text-sm font-medium tabular-nums">
   586→            {product.rating.toFixed(1)}
   587→          </span>
   588→          <span className="text-xs text-muted-foreground">
   589→            ({product.reviewCount})
   590→          </span>
   591→        </div>
   592→
   593→        {/* Cena */}
   594→        <div className="flex items-baseline gap-2">
   595→          <span className="text-lg font-bold text-foreground">
   596→            {formatPrice(product.price, product.currency)}
   597→          </span>
   598→          {product.compareAtPrice ? (
