   189→  const [loading, setLoading] = React.useState(true);
   190→  const [errorMsg, setErrorMsg] = React.useState<string | null>(null);
   191→  const [search, setSearch] = React.useState("");
   192→
   193→  const [formOpen, setFormOpen] = React.useState(false);
   194→  const [editing, setEditing] = React.useState<AdminListing | null>(null);
   195→  const [deleteId, setDeleteId] = React.useState<string | null>(null);
   196→  const [deleting, setDeleting] = React.useState(false);
   197→
   198→  const fetchListings = React.useCallback(async () => {
   199→    setLoading(true);
   200→    setErrorMsg(null);
   201→    try {
   202→      const res = await fetch("/api/admin/listings", {
   203→        headers: { "x-admin-password": adminPassword },
   204→      });
   205→      if (!res.ok) {
   206→        const d: unknown = await res.json();
   207→        const msg =
   208→          typeof d === "object" && d !== null && "error" in d
   209→            ? String((d as Record<string, unknown>).error)
   210→            : "Napaka pri pridobivanju lokalov";
   211→        setErrorMsg(msg);
   212→        return;
   213→      }

   216→        typeof data === "object" && data !== null && "listings" in data
   217→          ? (data as Record<string, unknown>).listings
   218→          : []) as AdminListing[];
   219→      setListings(list);
   220→    } catch (err) {
   221→      console.error("[admin/listings] fetch:", err);
   222→      setErrorMsg("Napaka pri povezavi s strežnikom");
   223→    } finally {
   224→      setLoading(false);
   225→    }
   226→  }, [adminPassword]);
   227→
   228→  React.useEffect(() => {
   229→    fetchListings();
   230→  }, [fetchListings]);
   231→
   232→  const filtered = React.useMemo(() => {
   233→    if (!search.trim()) return listings;

   235→    return listings.filter(
   236→      (l) =>
   237→        l.name.toLowerCase().includes(q) ||
   238→        (l.destinationName ?? "").toLowerCase().includes(q) ||
   239→        l.category.toLowerCase().includes(q)
   240→    );
   241→  }, [listings, search]);
   242→

   244→    setEditing(null);
   245→    setFormOpen(true);
   246→  };
   247→

   268→          typeof d === "object" && d !== null && "error" in d

   482→function LeadsTab({ adminPassword }: { adminPassword: string }) {
   483→  const [leads, setLeads] = React.useState<Lead[]>([]);
   484→  const [loading, setLoading] = React.useState(true);
   485→  const [errorMsg, setErrorMsg] = React.useState<string | null>(null);
   486→
   487→  const fetchLeads = React.useCallback(async () => {
   488→    setLoading(true);
   489→    setErrorMsg(null);
   490→    try {
   491→      const res = await fetch("/api/admin/leads", {
   492→        headers: { "x-admin-password": adminPassword },
   493→      });
   494→      if (!res.ok) {
   495→        const d: unknown = await res.json();
   496→        const msg =
   497→          typeof d === "object" && d !== null && "error" in d
   498→            ? String((d as Record<string, unknown>).error)
   499→            : "Napaka pri pridobivanju leadov";
   500→        setErrorMsg(msg);
   501→        return;
   502→      }
   503→      const data: unknown = await res.json();
   504→      const list = (
   505→        typeof data === "object" && data !== null && "leads" in data
   506→          ? (data as Record<string, unknown>).leads
   507→          : []) as Lead[];
   508→      setLeads(list);
   509→    } catch (err) {
   510→      console.error("[admin/leads] fetch:", err);
   511→      setErrorMsg("Napaka pri povezavi s strežnikom");
   512→    } finally {
   513→      setLoading(false);
   514→    }
   515→  }, [adminPassword]);
   516→
   517→  React.useEffect(() => {
   518→    fetchLeads();
   519→  }, [fetchLeads]);
   520→
   521→  const handleCycleStatus = async (lead: Lead) => {
   522→    const next = STATUS_NEXT[lead.status];
   523→    // Optimistic update
   524→    setLeads((prev) =>
   525→      prev.map((l) => (l.id === lead.id ? { ...l, status: next } : l))
   526→    );
   527→    try {
   528→      const res = await fetch("/api/admin/leads", {
   529→        method: "PUT",
   530→        headers: {
   531→          "Content-Type": "application/json",
   532→          "x-admin-password": adminPassword,
   533→        },
   534→        body: JSON.stringify({ id: lead.id, status: next }),
   535→      });
   536→      if (!res.ok) {
   537→        // Revert on failure
   538→        setLeads((prev) =>
   539→          prev.map((l) => (l.id === lead.id ? { ...l, status: lead.status } : l))
   540→        );
   541→        const d: unknown = await res.json();
   542→        const msg =
   543→          typeof d === "object" && d !== null && "error" in d
   544→            ? String((d as Record<string, unknown>).error)
   545→            : "Napaka pri posodabljanju";
   546→        setErrorMsg(msg);
   547→      }
   548→    } catch (err) {
   549→      console.error("[admin/leads] update:", err);
   550→      setLeads((prev) =>
   551→        prev.map((l) => (l.id === lead.id ? { ...l, status: lead.status } : l))
   552→      );
   553→      setErrorMsg("Napaka pri povezavi s strežnikom");
   554→    }
   555→  };
   556→
   557→  const handleExportCsv = () => {
   558→    if (leads.length === 0) return;
   559→    const headers = [
   560→      "Datum",
   561→      "Ime",
   562→      "Email",
   563→      "Telefon",
   564→      "Lokal",
   565→      "Tip",
   566→      "Kraj",
   567→      "Paket",
   568→      "Status",
   569→      "Sporočilo",
   570→    ];
   571→    const rows = leads.map((l) => [
   572→      new Date(l.timestamp).toLocaleString("sl-SI"),
   573→      l.name,
   574→      l.email,
   575→      l.phone ?? "",
   576→      l.businessName,
   577→      l.businessType,
   578→      l.location,
   579→      l.plan,
   580→      STATUS_LABELS[l.status],
   581→      (l.message ?? "").replace(/[\r\n]+/g, " "),
   582→    ]);
   583→    const csv = [headers, ...rows]
   584→      .map((row) =>
   585→        row
