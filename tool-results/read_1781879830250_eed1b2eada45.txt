    60→
    61→// Omejitve števila lokalov glede na paket in beta status
    62→const PLAN_LIMITS_NORMAL: Record<ListingPlan, number> = {
    63→  free: 1,
    64→  premium: 5,
    65→  enterprise: Infinity,
    66→};
    67→
    68→const PLAN_LIMITS_BETA: Record<ListingPlan, number> = {
    69→  free: 3,
    70→  premium: 8,
    71→  enterprise: Infinity,
    72→};
    73→
    74→interface BetaStatus {
    75→  isActive: boolean;
    76→  listingCount: number;
    77→  remainingToMonetization: number;
    78→  message: string;
    79→  betaEndDate: string;
    80→}
    81→
    82→export default function OwnerDashboardPage() {
    83→  const router = useRouter();
    84→  const { toast } = useToast();
    85→  const { data: session, status } = useSession();
    86→
    87→  const [listings, setListings] = useState<Listing[]>([]);
    88→  const [loadingListings, setLoadingListings] = useState(true);
    89→  const [formOpen, setFormOpen] = useState(false);
    90→  const [editing, setEditing] = useState<Listing | null>(null);
    91→  const [deleteId, setDeleteId] = useState<string | null>(null);
    92→  const [deleting, setDeleting] = useState(false);
    93→
    94→  // Beta status (client-side fetch)
    95→  const [betaStatus, setBetaStatus] = useState<BetaStatus | null>(null);
    96→
    97→  // Redirect na prijavo če ni prijavljen
    98→  useEffect(() => {
    99→    if (status === "unauthenticated") {
   100→      router.push("/owner/prijava");
   101→    }
   102→  }, [status, router]);
   103→
   104→  // Fetch beta status
   105→  useEffect(() => {
   106→    fetch("/api/beta-status")
   107→      .then((r) => r.json())
   108→      .then((d: BetaStatus) => setBetaStatus(d))
   109→      .catch(() => {});
   110→  }, []);
   111→
   112→  const fetchListings = useCallback(async () => {
   113→    setLoadingListings(true);
   114→    try {
   115→      const res = await fetch("/api/owner/listings", { cache: "no-store" });
   116→      if (!res.ok) throw new Error();
   117→      const data = await res.json();
   118→      setListings(data.listings || []);
   119→    } catch {
   120→      toast({
   121→        variant: "destructive",
   122→        title: "Napaka",
   123→        description: "Ni mogoče naložiti lokalov.",
   124→      });
   125→    } finally {
   126→      setLoadingListings(false);
   127→    }
   128→  }, [toast]);
   129→
   130→  useEffect(() => {
   131→    if (status === "authenticated") {
   132→      fetchListings();
   133→    }
   134→  }, [status, fetchListings]);
   135→
   136→  const plan = (session?.user?.plan as ListingPlan) || "free";
   137→  const isBetaActive = betaStatus?.isActive ?? true;
   138→  const planLimits = isBetaActive ? PLAN_LIMITS_BETA : PLAN_LIMITS_NORMAL;
   139→  const planLimit = planLimits[plan];
   140→  const planLimitLabel =
   141→    planLimit === Infinity ? "neomejeno" : String(planLimit);
   142→  const listingsCount = listings.length;
   143→  const canAddMore =
   144→    planLimit === Infinity ? true : listingsCount < planLimit;
   145→
   146→  const handleAdd = () => {
   147→    if (!canAddMore) {
   148→      toast({
   149→        variant: "destructive",
   150→        title: "Dosežen limit",
   151→        description: `Vaš paket (${plan}) omogoča največ ${planLimitLabel} ${
   152→          planLimit === 1 ? "lokal" : "lokalov"
   153→        }. Nadgradite naročnino.`,
   154→      });
   155→      return;
   156→    }
   157→    setEditing(null);
   158→    setFormOpen(true);
   159→  };
   160→
   161→  const handleEdit = (listing: Listing) => {
   162→    setEditing(listing);
   163→    setFormOpen(true);
   164→  };
   165→
   166→  const handleDelete = async () => {
   167→    if (!deleteId) return;
   168→    setDeleting(true);
   169→    try {
   170→      const res = await fetch(`/api/owner/listings/${deleteId}`, {
   171→        method: "DELETE",
   172→      });
   173→      if (!res.ok) {
   174→        const data = await res.json().catch(() => ({}));
   175→        throw new Error(data?.error ?? "Brisanje ni uspelo.");
   176→      }
   177→      toast({
   178→        title: "Izbrisano",
   179→        description: "Lokal je bil uspešno izbrisan.",
   180→      });
   181→      setListings((prev) => prev.filter((l) => l.id !== deleteId));
   182→    } catch (err) {
   183→      toast({
   184→        variant: "destructive",
   185→        title: "Napaka",
   186→        description:
   187→          err instanceof Error ? err.message : "Brisanje ni uspelo.",
   188→      });
   189→    } finally {
   190→      setDeleting(false);
   191→      setDeleteId(null);
   192→    }
   193→  };
   194→
   195→  // Loading state
   196→  if (status === "loading") {
   197→    return (
   198→      <main className="min-h-screen flex items-center justify-center bg-muted/30">
   199→        <div className="flex flex-col items-center gap-3 text-muted-foreground">
   200→          <Loader2 className="size-8 animate-spin" aria-hidden="true" />
   201→          <p className="text-sm">Nalagam portal...</p>
   202→        </div>
   203→      </main>
   204→    );
   205→  }
   206→
   207→  // Unauthenticated — redirect se sproži v useEffect
   208→  if (status === "unauthenticated" || !session?.user) {
   209→    return null;
   210→  }
   211→
   212→  return (
   213→    <main className="min-h-screen bg-muted/30 flex flex-col">
   214→      {/* Beta banner na vrhu */}
   215→      <BetaBanner />
   216→
   217→      {/* Header */}
   218→      <header className="bg-background border-b border-border sticky top-0 z-30">
   219→        <div className="mx-auto max-w-6xl px-4 py-3 sm:py-4 flex items-center justify-between gap-3">
   220→          <div className="flex items-center gap-3 min-w-0">
   221→            <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-primary text-primary-foreground">
   222→              <Building2 className="size-5" aria-hidden="true" />
   223→            </div>
   224→            <div className="min-w-0">
   225→              <h1 className="text-base font-bold sm:text-lg leading-tight">
   226→                Moj portal
   227→              </h1>
   228→              <p className="text-xs text-muted-foreground truncate">
   229→                {session.user.businessName || session.user.name}
   230→              </p>
   231→            </div>
   232→          </div>
   233→          <div className="flex items-center gap-2 shrink-0">
   234→            <PlanBadge plan={plan} />
   235→            <Button
   236→              variant="outline"
   237→              size="sm"
   238→              onClick={() => signOut({ callbackUrl: "/owner/prijava" })}
   239→              className="gap-1.5"
   240→            >
   241→              <LogOut className="size-3.5" aria-hidden="true" />
   242→              <span className="hidden sm:inline">Odjava</span>
   243→            </Button>
   244→          </div>
   245→        </div>
   246→      </header>
   247→
   248→      {/* Content */}
   249→      <div className="mx-auto max-w-6xl w-full px-4 py-6 sm:py-8 flex-1">
   250→        <Tabs defaultValue="listings" className="w-full">
   251→          <TabsList className="grid w-full grid-cols-3 mb-6">
   252→            <TabsTrigger value="listings" className="gap-1.5 text-xs sm:text-sm">
   253→              <Building className="size-3.5" aria-hidden="true" />
   254→              <span className="hidden sm:inline">Moji lokalci</span>
   255→              <span className="sm:hidden">Lokalci</span>
   256→            </TabsTrigger>
   257→            <TabsTrigger value="narocnina" className="gap-1.5 text-xs sm:text-sm">
   258→              <Crown className="size-3.5" aria-hidden="true" />
   259→              <span className="hidden sm:inline">Naročnina</span>
   260→              <span className="sm:hidden">Naročnina</span>
   261→            </TabsTrigger>
   262→            <TabsTrigger value="statistika" className="gap-1.5 text-xs sm:text-sm">
   263→              <TrendingUp className="size-3.5" aria-hidden="true" />
   264→              <span className="hidden sm:inline">Statistika</span>
   265→              <span className="sm:hidden">Statistika</span>
   266→            </TabsTrigger>
   267→          </TabsList>
   268→
   269→          {/* TAB 1: Moji lokalci */}
   270→          <TabsContent value="listings" className="space-y-4">
   271→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   272→              <div>
   273→                <h2 className="text-lg font-bold sm:text-xl">
   274→                  Moji lokalci
   275→                </h2>
   276→                <p className="text-sm text-muted-foreground">
   277→                  {listingsCount} od {planLimitLabel} lokalov · paket{" "}
   278→                  {PLAN_LABELS[plan]}
   279→                </p>
   280→              </div>
   281→              <Button
   282→                onClick={handleAdd}
   283→                disabled={!canAddMore}
   284→                className="bg-primary text-primary-foreground hover:bg-primary/90 gap-1.5 font-semibold"
   285→              >
   286→                <Plus className="size-4" aria-hidden="true" />
   287→                Dodaj lokal
   288→              </Button>
   289→            </div>
   290→
   291→            {/* Beta info badge v listings tab */}
   292→            {isBetaActive && (
   293→              <div className="flex flex-col sm:flex-row sm:items-center gap-3 rounded-lg border border-amber-300/60 bg-amber-50 dark:bg-amber-950/20 p-4">
   294→                <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300">
   295→                  <Rocket className="size-5" aria-hidden="true" />
   296→                </div>
   297→                <div className="flex-1 min-w-0">
   298→                  <p className="text-sm font-semibold text-amber-900 dark:text-amber-200">
   299→                    Beta: {plan === "free" ? "3 lokalci brezplačno" : plan === "premium" ? "8 lokalcev brezplačno" : "neomejeno brezplačno"}
   300→                    {plan === "free" && " (običajno 1)"}
   301→                    {plan === "premium" && " (običajno 5)"}
   302→                  </p>
   303→                  <p className="text-xs text-amber-700/80 dark:text-amber-300/70 mt-0.5">
   304→                    Med beta obdobjem so vsi paketi brezplačni. Vaš paket{" "}
   305→                    {PLAN_LABELS[plan]} vam omogoča {planLimitLabel} lokalov.
   306→                  </p>
   307→                </div>
   308→              </div>
   309→            )}
   310→
   311→            {!canAddMore && (
   312→              <Alert className="border-amber-400/50 bg-amber-50 dark:bg-amber-950/20">
   313→                <Crown className="size-4 text-amber-600" aria-hidden="true" />
   314→                <AlertTitle>Dosežen limit paketa</AlertTitle>
   315→                <AlertDescription>
   316→                  Vaš paket ({PLAN_LABELS[plan]}) omogoča največ{" "}
   317→                  {planLimitLabel}{" "}
   318→                  {planLimit === 1 ? "lokal" : "lokalov"}. Za več lokalov
   319→                  nadgradite na višji paket v zavihku &laquo;Naročnina&raquo;.
   320→                </AlertDescription>
   321→              </Alert>
   322→            )}
   323→
   324→            {loadingListings ? (
   325→              <div className="flex items-center justify-center py-16">
   326→                <Loader2
   327→                  className="size-8 animate-spin text-muted-foreground"
   328→                  aria-hidden="true"
   329→                />
   330→              </div>
   331→            ) : listings.length === 0 ? (
   332→              <EmptyState onAdd={handleAdd} canAdd={canAddMore} />
   333→            ) : (
   334→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   335→                {listings.map((listing) => (
   336→                  <ListingCard
   337→                    key={listing.id}
   338→                    listing={listing}
   339→                    onEdit={() => handleEdit(listing)}
   340→                    onDelete={() => setDeleteId(listing.id)}
   341→                  />
   342→                ))}
   343→              </div>
   344→            )}
   345→          </TabsContent>
   346→
   347→          {/* TAB 2: Naročnina */}
   348→          <TabsContent value="narocnina" className="space-y-6">
   349→            <SubscriptionTab
   350→              plan={plan}
   351→              session={session}
   352→              betaStatus={betaStatus}
   353→              onUpgraded={fetchListings}
   354→            />
   355→          </TabsContent>
   356→
   357→          {/* TAB 3: Statistika */}
   358→          <TabsContent value="statistika" className="space-y-6">
   359→            <StatisticsTab listings={listings} loading={loadingListings} />
   360→          </TabsContent>
   361→        </Tabs>
   362→      </div>
   363→
   364→      {/* Listing form dialog */}
   365→      <ListingFormDialog
   366→        open={formOpen}
   367→        onOpenChange={setFormOpen}
   368→        listing={editing}
   369→        onSaved={fetchListings}
   370→      />
   371→
   372→      {/* Delete confirmation */}
   373→      <AlertDialog
   374→        open={deleteId !== null}
   375→        onOpenChange={(open) => !open && setDeleteId(null)}
   376→      >
   377→        <AlertDialogContent>
   378→          <AlertDialogHeader>
   379→            <AlertDialogTitle>Izbriši lokal?</AlertDialogTitle>
   380→            <AlertDialogDescription>
   381→              To dejanje je nepovratno. Lokal bo trajno odstranjen iz portala
   382→              in ne bo več prikazan obiskovalcem.
   383→            </AlertDialogDescription>
   384→          </AlertDialogHeader>
   385→          <AlertDialogFooter>
   386→            <AlertDialogCancel disabled={deleting}>Prekliči</AlertDialogCancel>
   387→            <AlertDialogAction
   388→              onClick={handleDelete}
   389→              disabled={deleting}
   390→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90 gap-1.5"
   391→            >
   392→              {deleting ? (
   393→                <>
   394→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   395→                  Brišem...
   396→                </>
   397→              ) : (
   398→                <>
   399→                  <Trash2 className="size-4" aria-hidden="true" />
   400→                  Izbriši
   401→                </>
   402→              )}
   403→            </AlertDialogAction>
   404→          </AlertDialogFooter>
   405→        </AlertDialogContent>
   406→      </AlertDialog>
   407→    </main>
   408→  );
   409→}

   411→/* ====================== PLAN BADGE ====================== */
   412→
   413→function PlanBadge({ plan }: { plan: ListingPlan }) {
   414→  if (plan === "premium") {
   415→    return (
   416→      <Badge className="bg-amber-400 text-amber-950 hover:bg-amber-400 border-0 gap-1">
   417→        <Star className="size-3 fill-amber-950" aria-hidden="true" />
   418→        <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   419→      </Badge>
   420→    );
   421→  }
   422→  if (plan === "enterprise") {
   423→    return (
   424→      <Badge className="bg-primary text-primary-foreground hover:bg-primary border-0 gap-1">
   425→        <Crown className="size-3" aria-hidden="true" />
   426→        <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   427→      </Badge>
   428→    );
   429→  }
   430→  return (
   431→    <Badge variant="secondary" className="gap-1">
   432→      <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   433→      <span className="sm:hidden">Free</span>
   434→    </Badge>
   435→  );
   436→}

   477→  onEdit,
   478→  onDelete,
   479→}: {
   480→  listing: Listing;
   481→  onEdit: () => void;
   482→  onDelete: () => void;
   483→}) {

   494→          />

   499→            </span>

   503→        <Badge className="absolute left-2 top-2 bg-primary text-primary-foreground shadow-sm text-xs">

   510→              <Star className="size-3 fill-amber-950" aria-hidden="true" />
   511→              Izpost.
   512→            </Badge>
   513→          )}
   514→          {listing.verified && (
   515→            <Badge className="bg-primary text-primary-foreground border-0 text-xs">

   518→            </Badge>

   523→      {/* Vsebina */}

   528→          </h3>
   529→          {listing.destinationName && (
   530→            <p className="text-xs text-muted-foreground flex items-center gap-1">
   531→              <MapPin className="size-3" aria-hidden="true" />
   532→              {listing.destinationName}
   533→            </p>

   535→        </div>

   538→          {listing.description}
   539→        </p>
   540→
   541→        {/* Statistika */}
   542→        <div className="grid grid-cols-2 gap-2 mt-auto">
   543→          <div className="rounded-md border border-border/60 bg-muted/40 px-2 py-1.5">
   544→            <div className="flex items-center gap-1 text-[10px] uppercase tracking-wide text-muted-foreground">
   545→              <Eye className="size-3" aria-hidden="true" />
   546→              Ogledi
   547→            </div>
   548→            <div className="text-sm font-semibold tabular-nums">
   549→              {listing.viewCount.toLocaleString("sl-SI")}
   550→            </div>
   551→          </div>
   552→          <div className="rounded-md border border-border/60 bg-muted/40 px-2 py-1.5">
   553→            <div className="flex items-center gap-1 text-[10px] uppercase tracking-wide text-muted-foreground">
   554→              <MousePointerClick className="size-3" aria-hidden="true" />

   811→function SubscriptionCard({
   812→  plan,
   813→  current,
   814→  isBetaActive,
   815→  onUpgraded,
   816→}: {
   817→  plan: PricingPlan;
   818→  current: boolean;
   819→  isBetaActive: boolean;
   820→  onUpgraded: () => void;
   821→}) {
   822→  const isHighlighted = plan.highlighted;
   823→  const { toast } = useToast();
   824→  const [upgrading, setUpgrading] = useState(false);
   825→
   826→  const handleUpgrade = async () => {
   827→    if (current) return;
   828→    if (plan.id === "free") {
   829→      toast({
   830→        title: "Ni mogoče",
   831→        description: "Na free paket ne morete nadgraditi.",
   832→      });
   833→      return;
   834→    }
   835→
   836→    setUpgrading(true);
   837→    try {
   838→      const res = await fetch("/api/stripe/checkout", {
   839→        method: "POST",
   840→        headers: { "Content-Type": "application/json" },
   841→        body: JSON.stringify({ plan: plan.id }),
   842→      });
   843→      const data: unknown = await res.json();
   844→      if (!res.ok) {
   845→        const msg =
   846→          typeof data === "object" && data !== null && "error" in data
   847→            ? String((data as Record<string, unknown>).error)
   848→            : "Napaka pri nadgradnji.";
   849→        throw new Error(msg);
   850→      }
   851→      toast({
   852→        title: "Nadgrajeni!",
   853→        description: `Vaš paket je zdaj ${plan.name} ${
   854→          isBetaActive ? "(BREZPLAČNO med beta)" : ""
   855→        }.`,
   856→      });
   857→      // Osveži listings in session
   858→      onUpgraded();
   859→      // Reload page da se session refresh-a
   860→      window.location.reload();
   861→    } catch (err) {
   862→      toast({
   863→        variant: "destructive",
   864→        title: "Napaka",
   865→        description:
   866→          err instanceof Error ? err.message : "Napaka pri nadgradnji.",
   867→      });
   868→    } finally {
   869→      setUpgrading(false);
   870→    }
   871→  };
   872→
   873→  const hasOriginalPrice =
   874→    plan.betaFree === true &&
   875→    typeof plan.originalPrice === "number" &&
   876→    plan.originalPrice > 0;
   877→
   878→  return (
   879→    <div className={cn("relative flex", isHighlighted && "md:-mt-2 md:mb-2")}>
   880→      <Card
   881→        className={cn(
   882→          "w-full flex flex-col",
   883→          isHighlighted
   884→            ? "border-primary border-2 shadow-lg ring-1 ring-primary/20 md:scale-[1.02] z-10"
   885→            : "border-border",
   886→          current && "ring-2 ring-primary/40"
   887→        )}

   889→        {isHighlighted && plan.badge && (
   890→          <div className="absolute -top-3 left-1/2 -translate-x-1/2 z-20">
   891→            <Badge className="bg-amber-400 text-amber-950 hover:bg-amber-400 shadow-md border-0 font-semibold">
   892→              <Rocket className="size-3 mr-1" />
   893→              {plan.badge}
   894→            </Badge>
   895→          </div>
   896→        )}
   897→        {current && (
   898→          <div className="absolute -top-3 right-3 z-20">
   899→            <Badge className="bg-primary text-primary-foreground hover:bg-primary border-0 shadow-md">
   900→              <Check className="size-3 mr-1" aria-hidden="true" />
   901→              Trenutni
   902→            </Badge>
   903→          </div>
   904→        )}
   905→
   906→        <CardHeader className={cn(isHighlighted && "pt-7")}>
   907→          <CardTitle className="text-lg">{plan.name}</CardTitle>
   908→          <div className="mt-1">
   909→            {hasOriginalPrice ? (
   910→              <div className="space-y-1">
   911→                <div className="flex items-baseline gap-2">
   912→                  <span
   913→                    className="text-base text-muted-foreground line-through tabular-nums"
   914→                    aria-label={`Originalna cena ${plan.originalPrice} evrov na mesec`}
   915→                  >
   916→                    €{plan.originalPrice}/mes
   917→                  </span>
   918→                  <span className="text-sm font-semibold text-primary">
   919→                    Brezplačno med beta
   920→                  </span>
   921→                </div>
   922→              </div>
   923→            ) : plan.monthlyPrice > 0 ? (
   924→              <div className="flex items-baseline gap-1">
