   186→   186→// === TAB 1: LISTINGS ===
   187→   187→function ListingsTab({ adminPassword }: { adminPassword: string }) {
   188→   188→  const [listings, setListings] = React.useState<AdminListing[]>([]);
   189→   189→  const [loading, setLoading] = React.useState(true);
   190→   190→  const [errorMsg, setErrorMsg] = React.useState<string | null>(null);
   191→   191→  const [search, setSearch] = React.useState("");
   192→   192→
   193→   193→  const [formOpen, setFormOpen] = React.useState(false);
   194→   194→  const [editing, setEditing] = React.useState<AdminListing | null>(null);
   195→   195→  const [deleteId, setDeleteId] = React.useState<string | null>(null);
   196→   196→  const [deleting, setDeleting] = React.useState(false);
   197→   197→
   198→   198→  const fetchListings = React.useCallback(async () => {
   199→   199→    setLoading(true);
   200→   200→    setErrorMsg(null);
   201→   201→    try {
   202→   202→      const res = await fetch("/api/admin/listings", {
   203→   203→        headers: { "x-admin-password": adminPassword },
   204→   204→      });
   205→   205→      if (!res.ok) {
   206→   206→        const d: unknown = await res.json();
   207→   207→        const msg =
   208→   208→          typeof d === "object" && d !== null && "error" in d
   209→   209→            ? String((d as Record<string, unknown>).error)
   210→   210→            : "Napaka pri pridobivanju lokalov";
   211→   211→        setErrorMsg(msg);
   212→   212→        return;
   213→   213→      }
   214→   214→      const data: unknown = await res.json();
   215→   215→      const list = (
   216→   216→        typeof data === "object" && data !== null && "listings" in data
   217→   217→          ? (data as Record<string, unknown>).listings
   218→   218→          : []) as AdminListing[];
   219→   219→      setListings(list);
   220→   220→    } catch (err) {
   221→   221→      console.error("[admin/listings] fetch:", err);
   222→   222→      setErrorMsg("Napaka pri povezavi s strežnikom");
   223→   223→    } finally {
   224→   224→      setLoading(false);
   225→   225→    }
   226→   226→  }, [adminPassword]);
   227→   227→
   228→   228→  React.useEffect(() => {
   229→   229→    fetchListings();
   230→   230→  }, [fetchListings]);
   231→   231→
   232→   232→  const filtered = React.useMemo(() => {
   233→   233→    if (!search.trim()) return listings;
   234→   234→    const q = search.toLowerCase();
   235→   235→    return listings.filter(
   236→   236→      (l) =>
   237→   237→        l.name.toLowerCase().includes(q) ||
   238→   238→        (l.destinationName ?? "").toLowerCase().includes(q) ||
   239→   239→        l.category.toLowerCase().includes(q)
   240→   240→    );
   241→   241→  }, [listings, search]);
   242→   242→
   243→   243→  const handleOpenCreate = () => {
   244→   244→    setEditing(null);
   245→   245→    setFormOpen(true);
   246→   246→  };
   247→   247→
   248→   248→  const handleOpenEdit = (listing: AdminListing) => {
   249→   249→    setEditing(listing);
   250→   250→    setFormOpen(true);
   251→   251→  };
   252→   252→
   253→   253→  const handleSaved = () => {
   254→   254→    fetchListings();
   255→   255→  };
   256→   256→
   257→   257→  const handleConfirmDelete = async () => {
   258→   258→    if (!deleteId) return;
   259→   259→    setDeleting(true);
   260→   260→    try {
   261→   261→      const res = await fetch(`/api/admin/listings/${deleteId}`, {
   262→   262→        method: "DELETE",
   263→   263→        headers: { "x-admin-password": adminPassword },
   264→   264→      });
   265→   265→      if (!res.ok) {
   266→   266→        const d: unknown = await res.json();
   267→   267→        const msg =
   268→   268→          typeof d === "object" && d !== null && "error" in d
   269→   269→            ? String((d as Record<string, unknown>).error)
   270→   270→            : "Napaka pri brisanju";
   271→   271→        setErrorMsg(msg);
   272→   272→      } else {
   273→   273→        setListings((prev) => prev.filter((l) => l.id !== deleteId));
   274→   274→      }
   275→   275→    } catch (err) {
   276→   276→      console.error("[admin/listings] delete:", err);
   277→   277→      setErrorMsg("Napaka pri povezavi s strežnikom");
   278→   278→    } finally {
   279→   279→      setDeleting(false);
   280→   280→      setDeleteId(null);
   281→   281→    }
   282→   282→  };
   283→   283→
   284→   284→  return (
   285→   285→    <div className="space-y-4">
   286→   286→      {/* Toolbar */}
   287→   287→      <div className="flex flex-col sm:flex-row gap-3 sm:items-center sm:justify-between">
   288→   288→        <div>
   289→   289→          <h2 className="text-xl font-bold tracking-tight">Lokali</h2>
   290→   290→          <p className="text-sm text-muted-foreground">
   291→   291→            Upravljajte vse lokale na platformi.
   292→   292→          </p>
   293→   293→        </div>
   294→   294→        <div className="flex flex-col sm:flex-row gap-2 sm:items-center">
   295→   295→          <div className="relative">
   296→   296→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground pointer-events-none" />
   297→   297→            <Input
   298→   298→              value={search}
   299→   299→              onChange={(e) => setSearch(e.target.value)}
   300→   300→              placeholder="Iskanje po imenu..."
   301→   301→              className="pl-8 sm:w-64"
   302→   302→            />
   303→   303→          </div>
   304→   304→          <Button onClick={handleOpenCreate} className="shrink-0">
   305→   305→            <Plus className="size-4" />
   306→   306→            Nov lokal
   307→   307→          </Button>
   308→   308→        </div>
   309→   309→      </div>
   310→   310→
   311→   311→      {errorMsg && (
   312→   312→        <div className="flex items-start gap-2 rounded-md border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive">
   313→   313→          <AlertCircle className="size-4 shrink-0 mt-0.5" />
   314→   314→          <span>{errorMsg}</span>
   315→   315→        </div>
   316→   316→      )}
   317→   317→
   318→   318→      {/* Table */}
   319→   319→      <Card>
   320→   320→        <CardContent className="p-0">
   321→   321→          {loading ? (
   322→   322→            <div className="flex items-center justify-center gap-2 py-16 text-muted-foreground">
   323→   323→              <Loader2 className="size-4 animate-spin" />
   324→   324→              Nalaganje lokalov...
   325→   325→            </div>
   326→   326→          ) : filtered.length === 0 ? (
   327→   327→            <div className="py-16 text-center text-muted-foreground">
   328→   328→              <Building2 className="size-8 mx-auto mb-2 opacity-50" />
   329→   329→              {search ? "Ni najdenih lokalov." : "Ni še lokalov. Dodajte prvega."}
   330→   330→            </div>
   331→   331→          ) : (
   332→   332→            <div className="overflow-x-auto">
   333→   333→              <Table>
   334→   334→                <TableHeader>
   335→   335→                  <TableRow>
   336→   336→                    <TableHead>Ime</TableHead>
   337→   337→                    <TableHead className="hidden md:table-cell">Kategorija</TableHead>
   338→   338→                    <TableHead className="hidden lg:table-cell">Destinacija</TableHead>
   339→   339→                    <TableHead>Paket</TableHead>
   340→   340→                    <TableHead className="hidden sm:table-cell">Status</TableHead>
   341→   341→                    <TableHead className="hidden md:table-cell">Ocena</TableHead>
   342→   342→                    <TableHead className="hidden lg:table-cell">Ogledi</TableHead>
   343→   343→                    <TableHead className="text-right">Akcije</TableHead>
   344→   344→                  </TableRow>
   345→   345→                </TableHeader>
   346→   346→                <TableBody>
   347→   347→                  {filtered.map((l) => (
   348→   348→                    <TableRow key={l.id}>
   349→   349→                      <TableCell>
   350→   350→                        <div className="font-medium">{l.name}</div>
   351→   351→                        <div className="text-xs text-muted-foreground">
   352→   352→                          /{l.slug}
   353→   353→                        </div>
   354→   354→                      </TableCell>
   355→   355→                      <TableCell className="hidden md:table-cell">
   356→   356→                        <Badge variant="secondary">
   357→   357→                          {CATEGORY_LABELS[l.category as ListingCategory] ?? l.category}
   358→   358→                        </Badge>
   359→   359→                      </TableCell>
   360→   360→                      <TableCell className="hidden lg:table-cell text-muted-foreground">
   361→   361→                        {l.destinationName ?? "—"}
   362→   362→                      </TableCell>
   363→   363→                      <TableCell>
   364→   364→                        <PlanBadge plan={l.plan as ListingPlan} />
   365→   365→                      </TableCell>
   366→   366→                      <TableCell className="hidden sm:table-cell">
   367→   367→                        <div className="flex items-center gap-1">
   368→   368→                          {l.featured && (

   370→   370→                              <Star className="size-3 mr-0.5 fill-amber-500 text-amber-500" />
   371→   371→                              Izpost.
   372→   372→                            </Badge>
   373→   373→                          )}
   374→   374→                          {l.verified && (
   375→   375→                            <Badge variant="outline" className="border-emerald-300 bg-emerald-50 text-emerald-900">
   376→   376→                              <Check className="size-3 mr-0.5" />
   377→   377→                              Overjeno
   378→   378→                            </Badge>
   379→   379→                          )}
   380→   380→                          {!l.featured && !l.verified && (
   381→   381→                            <span className="text-xs text-muted-foreground">—</span>
   382→   382→                          )}
   383→   383→                        </div>
   384→   384→                      </TableCell>
   385→   385→                      <TableCell className="hidden md:table-cell">
   386→   386→                        <div className="flex items-center gap-1 text-sm">
   387→   387→                          <Star className="size-3.5 fill-amber-400 text-amber-400" />
   388→   388→                          <span className="font-medium">{l.rating.toFixed(1)}</span>
   389→   389→                          <span className="text-xs text-muted-foreground">({l.reviewCount})</span>

   391→   391→                      </TableCell>
   392→   392→                      <TableCell className="hidden lg:table-cell">
   393→   393→                        <div className="flex items-center gap-1 text-sm text-muted-foreground">
   394→   394→                          <Eye className="size-3.5" />
   395→   395→                          {l.viewCount}

   397→   397→                      </TableCell>
   398→   398→                      <TableCell className="text-right">
   399→   399→                        <div className="flex items-center justify-end gap-1">
   400→   400→                          <Button

   404→   404→                            aria-label={`Uredi ${l.name}`}
   405→   405→                          >

   415→   415→                          >
   416→   416→                            <Trash2 className="size-4" />
   417→   417→                            <span className="sr-only">Izbriši</span>
   418→   418→                          </Button>
   419→   419→                        </div>
   420→   420→                      </TableCell>
   421→   421→                    </TableRow>
   422→   422→                  ))}
   423→   423→                </TableBody>
   424→   424→              </Table>
   425→   425→            </div>
   426→   426→          )}
   427→   427→        </CardContent>
   428→   428→      </Card>
   429→   429→
   430→   430→      <div className="text-xs text-muted-foreground text-right">

   432→   432→      </div>
   433→   433→
   434→   434→      {/* Create/Edit Dialog */}
   435→   435→      <ListingForm
   436→   436→        open={formOpen}
   437→   437→        onOpenChange={setFormOpen}
   438→   438→        listing={editing}
   439→   439→        adminPassword={adminPassword}
   440→   440→        onSaved={handleSaved}
   441→   441→      />
   442→   442→
   443→   443→      {/* Delete confirmation */}
   444→   444→      <AlertDialog
   445→   445→        open={!!deleteId}
   446→   446→        onOpenChange={(o) => !o && setDeleteId(null)}
   447→   447→      >
   448→   448→        <AlertDialogContent>
   449→   449→          <AlertDialogHeader>
   450→   450→            <AlertDialogTitle>Izbriši lokal?</AlertDialogTitle>
   451→   451→            <AlertDialogDescription>
   452→   452→              Ta akcija je trajna. Lokal in vsi njegovi podatki bodo odstranjeni s platforme.
   453→   453→            </AlertDialogDescription>
   454→   454→          </AlertDialogHeader>
   455→   455→          <AlertDialogFooter>

   461→   461→            >
   462→   462→              {deleting ? (
   463→   463→                <>
   464→   464→                  <Loader2 className="size-4 animate-spin" />
   465→   465→                  Brisanje...
   466→   466→                </>
   467→   467→              ) : (
   468→   468→                <>
   469→   469→                  <Trash2 className="size-4" />
   470→   470→                  Izbriši
   471→   471→                </>
   472→   472→              )}
   473→   473→            </AlertDialogAction>
   474→   474→          </AlertDialogFooter>
   475→   475→        </AlertDialogContent>
   476→   476→      </AlertDialog>

   478→   478→  );
