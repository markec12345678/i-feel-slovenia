     4→import { useRouter } from "next/navigation";
     5→import { useSession, signOut } from "next-auth/react";
     6→import {

     9→  Plus,

    23→  ArrowRight,
    24→  ShieldCheck,

    26→} from "lucide-react";
    27→
    28→import { Button } from "@/components/ui/button";
    29→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    30→import { Badge } from "@/components/ui/badge";
    31→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    32→import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
    33→import {
    34→  AlertDialog,
    35→  AlertDialogAction,
    36→  AlertDialogCancel,
    37→  AlertDialogContent,
    38→  AlertDialogDescription,
    39→  AlertDialogFooter,
    40→  AlertDialogHeader,
    41→  AlertDialogTitle,
    42→} from "@/components/ui/alert-dialog";
    43→import { useToast } from "@/hooks/use-toast";
    44→import { cn } from "@/lib/utils";
    45→import { ListingFormDialog } from "@/components/owner/listing-form";
    46→import {
    47→  CATEGORY_LABELS,
    48→  CATEGORY_ICONS,
    49→  PLAN_LABELS,
    50→  type Listing,
    51→  type ListingPlan,
    52→} from "@/lib/listings-types";
    53→import { PRICING_PLANS, type PricingPlan } from "@/lib/pricing";
    54→
    55→const PLAN_LIMITS: Record<ListingPlan, number> = {
    56→  free: 1,
    57→  premium: 5,
    58→  enterprise: Infinity,
    59→};
    60→
    61→export default function OwnerDashboardPage() {
    62→  const router = useRouter();
    63→  const { toast } = useToast();
    64→  const { data: session, status } = useSession();
    65→
    66→  const [listings, setListings] = useState<Listing[]>([]);
    67→  const [loadingListings, setLoadingListings] = useState(true);
    68→  const [formOpen, setFormOpen] = useState(false);
    69→  const [editing, setEditing] = useState<Listing | null>(null);
    70→  const [deleteId, setDeleteId] = useState<string | null>(null);
    71→  const [deleting, setDeleting] = useState(false);
    72→
    73→  // Redirect na prijavo če ni prijavljen
    74→  useEffect(() => {
    75→    if (status === "unauthenticated") {
    76→      router.push("/owner/prijava");
    77→    }
    78→  }, [status, router]);
    79→
    80→  const fetchListings = useCallback(async () => {
    81→    setLoadingListings(true);
    82→    try {
    83→      const res = await fetch("/api/owner/listings", { cache: "no-store" });
    84→      if (!res.ok) throw new Error();
    85→      const data = await res.json();
    86→      setListings(data.listings || []);
    87→    } catch {
    88→      toast({
    89→        variant: "destructive",
    90→        title: "Napaka",
    91→        description: "Ni mogoče naložiti lokalov.",
    92→      });
    93→    } finally {
    94→      setLoadingListings(false);
    95→    }
    96→  }, [toast]);
    97→
    98→  useEffect(() => {
    99→    if (status === "authenticated") {
   100→      fetchListings();
   101→    }
   102→  }, [status, fetchListings]);
   103→
   104→  const plan = (session?.user?.plan as ListingPlan) || "free";
   105→  const planLimit = PLAN_LIMITS[plan];
   106→  const planLimitLabel =
   107→    planLimit === Infinity ? "neomejeno" : String(planLimit);
   108→  const listingsCount = listings.length;
   109→  const canAddMore =
   110→    planLimit === Infinity ? true : listingsCount < planLimit;
   111→
   112→  const handleAdd = () => {
   113→    if (!canAddMore) {
   114→      toast({
   115→        variant: "destructive",
   116→        title: "Dosežen limit",
   117→        description: `Vaš paket (${plan}) omogoča največ ${planLimitLabel} ${
   118→          planLimit === 1 ? "lokal" : "lokalov"
   119→        }. Nadgradite naročnino.`,
   120→      });
   121→      return;
   122→    }
   123→    setEditing(null);
   124→    setFormOpen(true);
   125→  };
   126→
   127→  const handleEdit = (listing: Listing) => {
   128→    setEditing(listing);
   129→    setFormOpen(true);
   130→  };
   131→
   132→  const handleDelete = async () => {
   133→    if (!deleteId) return;
   134→    setDeleting(true);
   135→    try {
   136→      const res = await fetch(`/api/owner/listings/${deleteId}`, {
   137→        method: "DELETE",
   138→      });
   139→      if (!res.ok) {
   140→        const data = await res.json().catch(() => ({}));
   141→        throw new Error(data?.error ?? "Brisanje ni uspelo.");
   142→      }
   143→      toast({
   144→        title: "Izbrisano",
   145→        description: "Lokal je bil uspešno izbrisan.",
   146→      });
   147→      setListings((prev) => prev.filter((l) => l.id !== deleteId));
   148→    } catch (err) {
   149→      toast({
   150→        variant: "destructive",
   151→        title: "Napaka",
   152→        description:
   153→          err instanceof Error ? err.message : "Brisanje ni uspelo.",
   154→      });
   155→    } finally {
   156→      setDeleting(false);
   157→      setDeleteId(null);
   158→    }
   159→  };
   160→
   161→  // Loading state
   162→  if (status === "loading") {
   163→    return (
   164→      <main className="min-h-screen flex items-center justify-center bg-muted/30">
   165→        <div className="flex flex-col items-center gap-3 text-muted-foreground">
   166→          <Loader2 className="size-8 animate-spin" aria-hidden="true" />
   167→          <p className="text-sm">Nalagam portal...</p>
   168→        </div>
   169→      </main>
   170→    );
   171→  }
   172→
   173→  // Unauthenticated — redirect se sproži v useEffect
   174→  if (status === "unauthenticated" || !session?.user) {
   175→    return null;
   176→  }
   177→
   178→  return (
   179→    <main className="min-h-screen bg-muted/30 flex flex-col">
   180→      {/* Header */}
   181→      <header className="bg-background border-b border-border sticky top-0 z-30">
   182→        <div className="mx-auto max-w-6xl px-4 py-3 sm:py-4 flex items-center justify-between gap-3">
   183→          <div className="flex items-center gap-3 min-w-0">
   184→            <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-primary text-primary-foreground">
   185→              <Building2 className="size-5" aria-hidden="true" />
   186→            </div>
   187→            <div className="min-w-0">
   188→              <h1 className="text-base font-bold sm:text-lg leading-tight">
   189→                Moj portal
   190→              </h1>
   191→              <p className="text-xs text-muted-foreground truncate">
   192→                {session.user.businessName || session.user.name}
   193→              </p>
   194→            </div>
   195→          </div>
   196→          <div className="flex items-center gap-2 shrink-0">
   197→            <PlanBadge plan={plan} />
   198→            <Button
   199→              variant="outline"
   200→              size="sm"
   201→              onClick={() => signOut({ callbackUrl: "/owner/prijava" })}
   202→              className="gap-1.5"
   203→            >
   204→              <LogOut className="size-3.5" aria-hidden="true" />
   205→              <span className="hidden sm:inline">Odjava</span>
   206→            </Button>
   207→          </div>
   208→        </div>
   209→      </header>
   210→
   211→      {/* Content */}
   212→      <div className="mx-auto max-w-6xl w-full px-4 py-6 sm:py-8 flex-1">
   213→        <Tabs defaultValue="listings" className="w-full">
   214→          <TabsList className="grid w-full grid-cols-3 mb-6">
   215→            <TabsTrigger value="listings" className="gap-1.5 text-xs sm:text-sm">
   216→              <Building className="size-3.5" aria-hidden="true" />
   217→              <span className="hidden sm:inline">Moji lokalci</span>
   218→              <span className="sm:hidden">Lokalci</span>
   219→            </TabsTrigger>
   220→            <TabsTrigger value="narocnina" className="gap-1.5 text-xs sm:text-sm">
   221→              <Crown className="size-3.5" aria-hidden="true" />
   222→              <span className="hidden sm:inline">Naročnina</span>
   223→              <span className="sm:hidden">Naročnina</span>
   224→            </TabsTrigger>
   225→            <TabsTrigger value="statistika" className="gap-1.5 text-xs sm:text-sm">
   226→              <TrendingUp className="size-3.5" aria-hidden="true" />
   227→              <span className="hidden sm:inline">Statistika</span>
   228→              <span className="sm:hidden">Statistika</span>
   229→            </TabsTrigger>
   230→          </TabsList>
   231→
   232→          {/* TAB 1: Moji lokalci */}
   233→          <TabsContent value="listings" className="space-y-4">
   234→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   235→              <div>
   236→                <h2 className="text-lg font-bold sm:text-xl">
   237→                  Moji lokalci
   238→                </h2>
   239→                <p className="text-sm text-muted-foreground">
   240→                  {listingsCount} od {planLimitLabel} lokalov · paket{" "}
   241→                  {PLAN_LABELS[plan]}
   242→                </p>
   243→              </div>
   244→              <Button
   245→                onClick={handleAdd}
   246→                disabled={!canAddMore}
   247→                className="bg-primary text-primary-foreground hover:bg-primary/90 gap-1.5 font-semibold"
   248→              >
   249→                <Plus className="size-4" aria-hidden="true" />
   250→                Dodaj lokal
   251→              </Button>
   252→            </div>
   253→
   254→            {!canAddMore && (
   255→              <Alert className="border-amber-400/50 bg-amber-50 dark:bg-amber-950/20">
   256→                <Crown className="size-4 text-amber-600" aria-hidden="true" />
   257→                <AlertTitle>Dosežen limit paketa</AlertTitle>
   258→                <AlertDescription>
   259→                  Vaš paket ({PLAN_LABELS[plan]}) omogoča največ{" "}
   260→                  {planLimitLabel}{" "}
   261→                  {planLimit === 1 ? "lokal" : "lokalov"}. Za več lokalov
   262→                  nadgradite na višji paket v zavihku &laquo;Naročnina&raquo;.
   263→                </AlertDescription>
   264→              </Alert>
   265→            )}
   266→
   267→            {loadingListings ? (
   268→              <div className="flex items-center justify-center py-16">
   269→                <Loader2
   270→                  className="size-8 animate-spin text-muted-foreground"
   271→                  aria-hidden="true"
   272→                />
   273→              </div>
   274→            ) : listings.length === 0 ? (
   275→              <EmptyState onAdd={handleAdd} canAdd={canAddMore} />
   276→            ) : (
   277→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   278→                {listings.map((listing) => (
   279→                  <ListingCard
   280→                    key={listing.id}
   281→                    listing={listing}
   282→                    onEdit={() => handleEdit(listing)}
   283→                    onDelete={() => setDeleteId(listing.id)}
   284→                  />
   285→                ))}
   286→              </div>
   287→            )}
   288→          </TabsContent>
   289→
   290→          {/* TAB 2: Naročnina */}
   291→          <TabsContent value="narocnina" className="space-y-6">
   292→            <SubscriptionTab plan={plan} session={session} />
   293→          </TabsContent>
   294→
   295→          {/* TAB 3: Statistika */}
   296→          <TabsContent value="statistika" className="space-y-6">
   297→            <StatisticsTab listings={listings} loading={loadingListings} />
   298→          </TabsContent>
   299→        </Tabs>
   300→      </div>
   301→
   302→      {/* Listing form dialog */}
   303→      <ListingFormDialog
   304→        open={formOpen}
   305→        onOpenChange={setFormOpen}
   306→        listing={editing}
   307→        onSaved={fetchListings}
   308→      />
   309→
   310→      {/* Delete confirmation */}
   311→      <AlertDialog
   312→        open={deleteId !== null}
   313→        onOpenChange={(open) => !open && setDeleteId(null)}
   314→      >
   315→        <AlertDialogContent>
   316→          <AlertDialogHeader>
   317→            <AlertDialogTitle>Izbriši lokal?</AlertDialogTitle>
   318→            <AlertDialogDescription>
   319→              To dejanje je nepovratno. Lokal bo trajno odstranjen iz portala
   320→              in ne bo več prikazan obiskovalcem.
   321→            </AlertDialogDescription>
   322→          </AlertDialogHeader>
   323→          <AlertDialogFooter>
   324→            <AlertDialogCancel disabled={deleting}>Prekliči</AlertDialogCancel>
   325→            <AlertDialogAction
   326→              onClick={handleDelete}
   327→              disabled={deleting}
   328→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90 gap-1.5"
   329→            >
   330→              {deleting ? (
   331→                <>
   332→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   333→                  Brišem...
   334→                </>
   335→              ) : (
   336→                <>
   337→                  <Trash2 className="size-4" aria-hidden="true" />
   338→                  Izbriši
   339→                </>
   340→              )}
   341→            </AlertDialogAction>
   342→          </AlertDialogFooter>
   343→        </AlertDialogContent>
   344→      </AlertDialog>
   345→    </main>
   346→  );
   347→}

   349→/* ====================== PLAN BADGE ====================== */
   350→
   351→function PlanBadge({ plan }: { plan: ListingPlan }) {
   352→  if (plan === "premium") {
   353→    return (
   354→      <Badge className="bg-amber-400 text-amber-950 hover:bg-amber-400 border-0 gap-1">
   355→        <Star className="size-3 fill-amber-950" aria-hidden="true" />
   356→        <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   357→      </Badge>
   358→    );
   359→  }
   360→  if (plan === "enterprise") {
   361→    return (
   362→      <Badge className="bg-primary text-primary-foreground hover:bg-primary border-0 gap-1">
   363→        <Crown className="size-3" aria-hidden="true" />
   364→        <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   365→      </Badge>
   366→    );
   367→  }
   368→  return (
   369→    <Badge variant="secondary" className="gap-1">
   370→      <span className="hidden sm:inline">{PLAN_LABELS[plan]}</span>
   371→      <span className="sm:hidden">Free</span>
   372→    </Badge>
   373→  );
   374→}

   527→/* ====================== SUBSCRIPTION TAB ====================== */
   528→
   529→interface SubscriptionTabProps {
   530→  plan: ListingPlan;
   531→  session: ReturnType<typeof useSession>["data"];
   532→}
   533→
   534→function SubscriptionTab({ plan, session }: SubscriptionTabProps) {
   535→  const subscriptionStatus =
   536→    (session?.user?.subscriptionStatus as string) || "none";
   537→  const isActive = subscriptionStatus === "active";
   538→
   539→  return (
   540→    <div className="space-y-6">
   541→      {/* Trenutni paket */}
   542→      <Card
   543→        className={cn(
   544→          "border-2",
   545→          plan === "free"
   546→            ? "border-border"
   547→            : plan === "premium"
   548→            ? "border-amber-400"
   549→            : "border-primary"
   550→        )}
   551→      >
   552→        <CardHeader>
   553→          <div className="flex items-start justify-between gap-3">
   554→            <div className="space-y-1">
   555→              <div className="flex items-center gap-2">
   556→                {plan === "premium" ? (
   557→                  <Star
   558→                    className="size-5 fill-amber-400 text-amber-400"
   559→                    aria-hidden="true"
   560→                  />
   561→                ) : plan === "enterprise" ? (
   562→                  <Crown className="size-5 text-primary" aria-hidden="true" />
   563→                ) : (
   564→                  <Building className="size-5 text-muted-foreground" aria-hidden="true" />
   565→                )}
   566→                <CardTitle className="text-xl">
   567→                  Trenutni paket: {PLAN_LABELS[plan]}
   568→                </CardTitle>
   569→              </div>
   570→              <p className="text-sm text-muted-foreground">
   571→                {plan === "free"
   572→                  ? "Brezplačni paket z 1 lokalom."
   573→                  : plan === "premium"
   574→                  ? "Premium paket z do 5 lokali in izpostavljenostjo."
   575→                  : "Enterprise paket z neomejenimi lokali in dodatnimi funkcijami."}
   576→              </p>
   577→            </div>
   578→            <PlanBadge plan={plan} />
   579→          </div>
   580→        </CardHeader>
   581→        {plan !== "free" && (
   582→          <CardContent className="pt-0 space-y-3">
   583→            <div className="flex items-center gap-2 text-sm">
   584→              <Badge
   585→                variant={isActive ? "default" : "secondary"}
   586→                className="gap-1"
   587→              >
   588→                <ShieldCheck className="size-3" aria-hidden="true" />
   589→                {subscriptionStatus === "active"
   590→                  ? "Aktivna"
   591→                  : subscriptionStatus === "canceled"
   592→                  ? "Preklicana"
   593→                  : subscriptionStatus === "past_due"
   594→                  ? "Zapadlo plačilo"
   595→                  : "Brez naročnine"}
   596→              </Badge>
   597→            </div>
   598→            <Button variant="outline" className="gap-1.5" disabled>
   599→              <CalendarClock className="size-4" aria-hidden="true" />
   600→              Upravljaj naročnino
   601→              <span className="text-xs text-muted-foreground ml-1">
   602→                (Stripe — kmalu)
   603→              </span>
   604→            </Button>
   605→          </CardContent>
   606→        )}
   607→      </Card>
   608→
   609→      {/* Pricing cards */}
   610→      <div>
   611→        <h3 className="text-lg font-bold mb-1">Nadgradi paket</h3>
   612→        <p className="text-sm text-muted-foreground mb-4">
   613→          Izberite paket, ki ustreza vašemu poslovanju.
   614→        </p>
   615→        <div className="grid grid-cols-1 gap-4 md:grid-cols-3 md:items-stretch">
   616→          {PRICING_PLANS.map((p) => (
   617→            <SubscriptionCard
   618→              key={p.id}
   619→              plan={p}
   620→              current={p.id === plan}
   621→            />
   622→          ))}
   623→        </div>
   624→      </div>
   625→    </div>
   626→  );
   627→}

   629→function SubscriptionCard({
   630→  plan,
   631→  current,
   632→}: {
   633→  plan: PricingPlan;
   634→  current: boolean;
   635→}) {
   636→  const isHighlighted = plan.highlighted;
   637→  const { toast } = useToast();
   638→
   639→  const handleUpgrade = () => {
   640→    if (current) return;
   641→    if (plan.id === "free") {
   642→      toast({
   643→        title: "Ni mogoče",
   644→        description: "Na free paket ne morete nadgraditi.",
   645→      });

   647→    }
   648→    toast({
   649→      title: "Stripe — kmalu na voljo",
   650→      description: `Nadgradnja na ${plan.name} paket bo kmalu omogočena preko Stripe.`,
   651→    });
   652→  };
   653→
   654→  return (
   655→    <div className={cn("relative flex", isHighlighted && "md:-mt-2 md:mb-2")}>
   656→      <Card
   657→        className={cn(
   658→          "w-full flex flex-col",
   659→          isHighlighted
   660→            ? "border-primary border-2 shadow-lg ring-1 ring-primary/20 md:scale-[1.02] z-10"
   661→            : "border-border",
   662→          current && "ring-2 ring-primary/40"

   670→            </Badge>
   671→          </div>
   672→        )}
   673→        {current && (
   674→          <div className="absolute -top-3 right-3 z-20">
   675→            <Badge className="bg-primary text-primary-foreground hover:bg-primary border-0 shadow-md">
   676→              <Check className="size-3 mr-1" aria-hidden="true" />
   677→              Trenutni
   678→            </Badge>
   679→          </div>
   680→        )}
   681→
   682→        <CardHeader className={cn(isHighlighted && "pt-7")}>
   683→          <CardTitle className="text-lg">{plan.name}</CardTitle>
   684→          <div className="mt-1">
   685→            {plan.monthlyPrice > 0 ? (
   686→              <div className="flex items-baseline gap-1">
   687→                <span className="text-3xl font-bold tabular-nums">
   688→                  €{plan.monthlyPrice}
   689→                </span>
   690→                <span className="text-sm text-muted-foreground">/mes</span>
   691→              </div>
   692→            ) : (
   693→              <span className="text-2xl font-bold text-muted-foreground">
   694→                Brezplačno
   695→              </span>
   696→            )}
   697→          </div>
   698→          <p className="text-xs text-muted-foreground mt-1">{plan.tagline}</p>
   699→        </CardHeader>
   700→
   701→        <CardContent className="flex flex-col gap-4 flex-1">
   702→          <ul className="space-y-2 flex-1">
   703→            {plan.features.map((feature, idx) => (
   704→              <li key={idx} className="flex items-start gap-2 text-xs">
   705→                <Check
   706→                  className="size-3.5 mt-0.5 shrink-0 text-primary"
   707→                  aria-hidden="true"
   708→                />
   709→                <span className="leading-relaxed">{feature}</span>
   710→              </li>
   711→            ))}
   712→          </ul>
   713→
   714→          <Button
   715→            onClick={handleUpgrade}
   716→            disabled={current}
   717→            variant={isHighlighted ? "default" : "outline"}
   718→            className={cn(
   719→              "w-full font-semibold",
   720→              isHighlighted
   721→                ? "bg-primary text-primary-foreground hover:bg-primary/90"
   722→                : "hover:bg-accent hover:text-accent-foreground",
   723→              current && "opacity-60 cursor-not-allowed"
   724→            )}

   727→              <>
   728→                <Check className="size-4 mr-1" aria-hidden="true" />
   729→                Trenutni paket
   730→              </>
   731→            ) : plan.id === "free" ? (
   732→              plan.cta
   733→            ) : (
   734→              <>
   735→                <Sparkles className="size-4 mr-1" aria-hidden="true" />

   737→                <ArrowRight className="size-4 ml-1" aria-hidden="true" />

   739→            )}
   740→          </Button>
   741→        </CardContent>
   742→      </Card>
   743→    </div>

   749→function StatisticsTab({
   750→  listings,
   751→  loading,
   752→}: {
   753→  listings: Listing[];
   754→  loading: boolean;
   755→}) {
   756→  if (loading) {
   757→    return (
   758→      <div className="flex items-center justify-center py-16">
   759→        <Loader2
   760→          className="size-8 animate-spin text-muted-foreground"
   761→          aria-hidden="true"
   762→        />
   763→      </div>
   764→    );
   765→  }
   766→
   767→  if (listings.length === 0) {
   768→    return (
   769→      <Card className="border-dashed">
   770→        <CardContent className="py-12 text-center">
   771→          <TrendingUp
   772→            className="size-10 mx-auto text-muted-foreground mb-3"
   773→            aria-hidden="true"
   774→          />
   775→          <h3 className="font-bold">Ni podatkov za prikaz</h3>
   776→          <p className="text-sm text-muted-foreground mt-1">
   777→            Dodajte vsaj en lokal za prikaz statistike.
   778→          </p>
   779→        </CardContent>
   780→      </Card>
   781→    );
   782→  }
   783→
   784→  const totalViews = listings.reduce((sum, l) => sum + l.viewCount, 0);
   785→  const totalClicks = listings.reduce((sum, l) => sum + l.clickCount, 0);
   786→  const conversion =
   787→    totalViews > 0 ? ((totalClicks / totalViews) * 100).toFixed(1) : "0.0";
   788→
   789→  // Top lokal po views
   790→  const topByViews = [...listings].sort(
   791→    (a, b) => b.viewCount - a.viewCount
   792→  )[0];
   793→
   794→  // Top 5 lokalov po klikih
   795→  const topByClicks = [...listings]
   796→    .sort((a, b) => b.clickCount - a.clickCount)
   797→    .slice(0, 5);
   798→
   799→  const maxClicks = Math.max(...topByClicks.map((l) => l.clickCount), 1);
   800→
   801→  return (
   802→    <div className="space-y-6">
   803→      {/* KPI kartice */}
   804→      <div className="grid grid-cols-2 gap-3 sm:gap-4 lg:grid-cols-4">
   805→        <KpiCard
   806→          icon={Eye}
   807→          label="Skupni ogledi"
   808→          value={totalViews.toLocaleString("sl-SI")}
   809→          color="primary"
   810→        />
   811→        <KpiCard
   812→          icon={MousePointerClick}
   813→          label="Skupni kliki"
   814→          value={totalClicks.toLocaleString("sl-SI")}
   815→          color="primary"
   816→        />
   817→        <KpiCard
   818→          icon={TrendingUp}
   819→          label="Konverzija"
   820→          value={`${conversion}%`}
   821→          color="amber"
   822→        />
   823→        <KpiCard
   824→          icon={Building}
   825→          label="Število lokalov"
   826→          value={String(listings.length)}
   827→          color="primary"
   828→        />
   829→      </div>
   830→
   831→      {/* Top lokal */}
   832→      {topByViews && (
   833→        <Card>
   834→          <CardHeader>
   835→            <CardTitle className="flex items-center gap-2 text-base">
   836→              <Star
   837→                className="size-4 fill-amber-400 text-amber-400"
   838→                aria-hidden="true"
   839→              />
   840→              Top lokal po ogledih
   841→            </CardTitle>
   842→          </CardHeader>
   843→          <CardContent>
   844→            <div className="flex items-center justify-between gap-3">
   845→              <div className="flex items-center gap-3 min-w-0">
   846→                <div className="flex size-10 items-center justify-center rounded-lg bg-primary/10 text-primary shrink-0">
   847→                  <span aria-hidden="true">
   848→                    {CATEGORY_ICONS[topByViews.category]}
   849→                  </span>
   850→                </div>
   851→                <div className="min-w-0">
   852→                  <p className="font-semibold truncate">{topByViews.name}</p>
   853→                  <p className="text-xs text-muted-foreground">
   854→                    {CATEGORY_LABELS[topByViews.category]}
   855→                    {topByViews.destinationName
   856→                      ? ` · ${topByViews.destinationName}`
   857→                      : ""}
   858→                  </p>
   859→                </div>
   860→              </div>
   861→              <div className="text-right shrink-0">
   862→                <div className="text-lg font-bold tabular-nums">
   863→                  {topByViews.viewCount.toLocaleString("sl-SI")}
   864→                </div>
   865→                <div className="text-xs text-muted-foreground">ogledov</div>
   866→              </div>
   867→            </div>
   868→          </CardContent>
   869→        </Card>
   870→      )}
   871→
   872→      {/* Top 5 po klikih — simple bar chart */}
   873→      <Card>
   874→        <CardHeader>
   875→          <CardTitle className="flex items-center gap-2 text-base">
   876→            <MousePointerClick className="size-4 text-primary" aria-hidden="true" />
   877→            Kliki po lokalah (top {topByClicks.length})
   878→          </CardTitle>
   879→        </CardHeader>
   880→        <CardContent className="space-y-3">
   881→          {topByClicks.map((l, idx) => {
   882→            const widthPct = maxClicks > 0 ? (l.clickCount / maxClicks) * 100 : 0;
   883→            return (
   884→              <div key={l.id} className="space-y-1">
   885→                <div className="flex items-center justify-between gap-2 text-sm">
   886→                  <span className="flex items-center gap-2 min-w-0">
   887→                    <span className="text-xs text-muted-foreground tabular-nums shrink-0">
   888→                      #{idx + 1}
   889→                    </span>
   890→                    <span className="truncate">{l.name}</span>
   891→                  </span>
   892→                  <span className="font-semibold tabular-nums shrink-0">
   893→                    {l.clickCount.toLocaleString("sl-SI")}
   894→                  </span>
   895→                </div>
   896→                <div className="h-2 w-full rounded-full bg-muted overflow-hidden">
   897→                  <div
   898→                    className="h-full rounded-full bg-primary transition-all"
   899→                    style={{ width: `${widthPct}%` }}
   900→                    role="progressbar"
   901→                    aria-valuenow={l.clickCount}
   902→                    aria-valuemin={0}
   903→                    aria-valuemax={maxClicks}
   904→                  />
   905→                </div>
   906→              </div>
   907→            );
   908→          })}
   909→        </CardContent>
   910→      </Card>
   911→
   912→      {/* Opomba o zgodovinski statistiki */}
   913→      <Alert>
   914→        <AlertCircle className="size-4" aria-hidden="true" />
   915→        <AlertTitle>Podrobna časovna statistika</AlertTitle>
   916→        <AlertDescription>
   917→          Prikazujemo skupne števce. Podrobna časovna statistika (po dnevih
   918→          zadnjih 30 dni) bo na voljo s paketoma Premium in Enterprise.
   919→        </AlertDescription>
   920→      </Alert>
   921→    </div>
   922→  );
   923→}
